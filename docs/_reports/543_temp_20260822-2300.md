### ⚠️ 為什麼 `status: "draft"`

```

```

而 `FAMILIES` 住在 `packages/shared/src/content/templates/expand.ts` —— **本 lane 柵欄外**
（⛔ 8 條 lane 併行，2007 行的共用檔）。
⇒ 標 `enabled` 會讓 `paramsSchema.test.ts` 對**每一條 lane**都紅。`draft` 是唯一誠實的值。

```ts
/** 三個【移動中的模型】家族共用的核心。⛔ 沒有任何一支技能的 if。 */
function modelFxCore(
  t: TemplateDoc,
  p: Record<string, unknown>,
  path: "forward" | "radial",
): Extract<EffectDef, { kind: "spawnModelFx" }> {
  return {
    kind: "spawnModelFx",
    shape: "single",
    modelKey: docRef(t, p, "modelKey"),
    path,
    speed: num(t, p, "speed"),
    distance: num(t, p, "distance"),
    ...(has(t, p, "spinDegPerSec") ? { spinDegPerSec: num(t, p, "spinDegPerSec") } : {}),
    ...(has(t, p, "scale") ? { scale: num(t, p, "scale") } : {}),
    touchRadius: num(t, p, "touchRadius"),
    touchSide: "enemies",
    onTouch: [
      {
        kind: "damage",
        damageType: damageType(t, p, "damageType"),
        amount: { damageTier: str(t, p, "touchDamageTier") as DamageTierName },
      },
    ],
  };
}
```
