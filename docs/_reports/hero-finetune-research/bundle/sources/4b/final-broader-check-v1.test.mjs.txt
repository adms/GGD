import fs from'node:fs';import path from'node:path';import test from'node:test';import assert from'node:assert/strict';import{fileURLToPath}from'node:url';import{scoreGroups,sliceArms}from'./final-broader-check-v1.mjs';import{verify}from'./final-regression-data-v1.mjs';
const root=path.resolve(path.dirname(fileURLToPath(import.meta.url)),'../../../outputs/forge-final-three-hours-20260906/final-broader-data-v1'),read=p=>JSON.parse(fs.readFileSync(p)),manifest=verify(root),cases=read(path.join(root,'cases.private.json'));
function fixtures(){return['base','r3','native','fused-bf16','mlx-8bit'].map((name,i)=>{const model=i<3?'/CPU-FIXTURE/base':'/CPU-FIXTURE/'+name,adapter=i===1||i===2?'/CPU-FIXTURE/adapter':null;return{name,model,adapter,raw:{complete:true,metadata:{modelPath:model,adapter,thinking:false,versions:{test:'CPU synthetic scoring only; no model output'},grammar:'fixture'},results:cases.map((c,j)=>({id:c.id,requestDigest:c.requestDigest,seed:20260906+j,value:c.target,error:null,seconds:0}))}};});}
test('broader scorer covers all 8 groups, five arms and all 582 cases without changing selection',()=>{const arms=fixtures(),rs=scoreGroups(cases,arms,manifest.groups);assert.equal(rs.length,8);assert.equal(rs.reduce((s,r)=>s+r.precisionComparison.scores.native.total,0),582);for(const r of rs){assert.equal(r.baseComparison.scores.base.passed,r.baseComparison.scores.base.total);assert(r.precisionComparison.comparisons.every(c=>c.preservation.pass));assert.equal(r.precisionComparison.releaseQualified,false);}assert.equal(sliceArms(arms,manifest.groups[0].ids)[0].raw.results.length,63);});
test('changed request digest, seed or an incomplete precision arm is rejected',()=>{for(const change of[a=>a[4].raw.results[0].requestDigest='wrong',a=>a[3].raw.results[0].seed++,a=>a[2].raw.complete=false]){const a=fixtures();change(a);assert.throws(()=>scoreGroups(cases,a,manifest.groups));}});
