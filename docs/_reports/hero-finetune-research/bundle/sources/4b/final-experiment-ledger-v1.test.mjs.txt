import test from'node:test';import assert from'node:assert/strict';import{trainingFacts,summarizeSelection,render}from'./final-experiment-ledger-v1.mjs';
test('partial training and missing receipts cannot become complete data exposure',()=>{const fixture={status:'partial',trainingComplete:false,totalSeconds:90,peakMetalBytes:3*1024**3,preparedExamples:550,recipe:{epochs:1},processedExamples:321,committedExamples:320,discardedPartialGradientExamples:1,optimizerSteps:80};const f=trainingFacts(fixture);assert.equal(f.complete,false);assert.equal(f.processedExamples,321);assert.equal(f.committedExamples,320);assert.equal(f.minutes,1.5);const unknown={...fixture};delete unknown.processedExamples;delete unknown.committedExamples;assert.equal(trainingFacts(unknown).committedExamples,null);assert.throws(()=>trainingFacts({recipe:{epochs:1},preparedExamples:550}));});
test('R3 control winning a later run is not reported as new learning',()=>{const s=summarizeSelection({selected:{step:0,label:'unchanged-r3',adapterSha256:'fixture'},testUsed:false},'R7');assert.equal(s.newTrainingSelected,false);assert.equal(s.testUsedForSelection,false);assert.equal(summarizeSelection(null,'R8'),null);const text=render({records:[{name:'R8',counts:{train:550,dev:179,test:228},stageStatus:'data-prepared-no-run-state',penalty:null,training:null,selection:null,note:'fixture only'}]});assert(text.includes('未測'));assert(text.includes('未選'));assert(!text.includes('訓練完成 |'));});
