import test from 'node:test';import assert from 'node:assert/strict';import {canonical as r6} from './r6-final-report-v2.mjs';import {canonical as r7} from './r7-final-report-v2.mjs';
test('report replay ignores only JSON-omitted optional fields, never real scores',()=>{for(const fn of [r6,r7]){const x={rows:[{id:'x',reason:undefined,score:{pass:false,unsafe:true}}]},stored=JSON.parse(JSON.stringify(x));assert.deepEqual(fn(x),stored);assert.throws(()=>assert.deepEqual(fn({...x,rows:[{...x.rows[0],score:{pass:true,unsafe:false}}]}),stored));assert.throws(()=>assert.deepEqual(fn({...x,rows:[{...x.rows[0],reason:'changed'}]}),stored));}});
