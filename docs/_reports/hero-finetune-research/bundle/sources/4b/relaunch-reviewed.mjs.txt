/** Copy only reviewed, immutable inputs into a new run; never copy model state. */
import fs from 'node:fs';import path from 'node:path';import crypto from 'node:crypto';
const [src,dst]=process.argv.slice(2);if(!src||!dst||!path.isAbsolute(src)||!path.isAbsolute(dst)||fs.existsSync(dst))throw Error('ABSOLUTE_SOURCE_AND_NEW_DESTINATION_REQUIRED');
const read=n=>JSON.parse(fs.readFileSync(path.join(src,n),'utf8'));const cases=read('cases.private.json'),review=read('semantic-review.json'),manifest=read('dataset-manifest.json');
const digest=crypto.createHash('sha256').update(JSON.stringify(cases)).digest('hex');if(review.status!=='approved-for-research-training'||review.casesSha256!==digest||manifest.casesSha256!==digest||review.reviewedRows!==cases.length)throw Error('REVIEW_MISMATCH');
const files=['cases.private.json','train.jsonl','train-requests.json','dev-requests.json','test-requests.json','dataset-manifest.json','semantic-review.json','catalog.json','experiment-policy.json','source-pins.json','curated-contract-check.json','QUESTION_REVIEW.md','QUALITY_REVIEW.md','legacy-quarantine.json','vfx-registry-quarantine.json'];
fs.mkdirSync(dst);for(const name of files){fs.copyFileSync(path.join(src,name),path.join(dst,name),fs.constants.COPYFILE_EXCL);if(!fs.readFileSync(path.join(src,name)).equals(fs.readFileSync(path.join(dst,name))))throw Error('COPY_MISMATCH');}
fs.writeFileSync(path.join(dst,'relaunch-receipt.json'),JSON.stringify({source:src,casesSha256:digest,files,baseRestart:'clean; no partial adapter or optimizer copied',deadline:read('experiment-policy.json').deadline,createdAt:new Date().toISOString()},null,2)+'\n',{flag:'wx'});console.log(JSON.stringify({destination:dst,counts:manifest.counts,deadline:read('experiment-policy.json').deadline}));
