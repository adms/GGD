import fs from'node:fs';import path from'node:path';import assert from'node:assert/strict';import{fileURLToPath}from'node:url';import{build}from'./r8-data-v1.mjs';import{fileHash}from'./precision-diagnostic.mjs';import{digest}from'./dataset.mjs';
const read=p=>JSON.parse(fs.readFileSync(p));
export function verify(root){
 const m=read(path.join(root,'DATA_PREPARATION.json'));for(const p of m.sourcePins)assert.equal(fileHash(p.path),p.sha256,'PIN_CHANGED:'+p.path);
 const data=build(read(path.join(m.parent,'cases.private.json')),read(path.join(m.community,'cases.private.json')),read(path.join(m.parent,'catalog.json'))),cases=read(path.join(root,'cases.private.json'));assert.deepEqual(cases,data.cases);assert.equal(digest(cases),m.casesSha256);assert.deepEqual(read(path.join(root,'catalog.json')),data.catalog);assert.deepEqual(read(path.join(root,'withdrawn-labels.json')),data.withdrawn);assert.deepEqual(read(path.join(root,'changed-inputs.json')),data.changes);
 for(const split of['train','dev','test']){const rows=cases.filter(c=>c.split===split);assert.deepEqual(read(path.join(root,split+'-requests.json')),rows.map(({id,messages,requestDigest})=>({id,messages,requestDigest})));const b=read(path.join(root,split+'-token-budget.json'));assert(b.allFit&&b.cpuOnly&&!b.thinking);assert.equal(b.requestsSha256,fileHash(path.join(root,split+'-requests.json')));assert.equal(b.count,rows.length);}
 assert.deepEqual(fs.readFileSync(path.join(root,'train.jsonl'),'utf8').trim().split('\n').map(JSON.parse),cases.filter(c=>c.split==='train').map(({id,messages,target})=>({id,messages,target})));
 return{verifiedAt:new Date().toISOString(),exactRebuild:true,counts:m.counts,oldSourceRowsUnchanged:data.oldSourceRowsUnchanged,changedInputCounts:m.changedInputCounts,allPromptsFit:true,withdrawnTrainingLabels:2,reviewedReplacementPositives:2,communityTrainingRows:84,trainingStarted:false,policyChosen:false,releaseQualified:false};
}
if(process.argv[1]&&path.resolve(process.argv[1])===fileURLToPath(import.meta.url)){const root=path.resolve(process.argv[2]),result=verify(root);fs.writeFileSync(path.join(root,'DATA_ADMISSION.json'),JSON.stringify(result,null,2)+'\n',{flag:'wx'});console.log(JSON.stringify(result));}
