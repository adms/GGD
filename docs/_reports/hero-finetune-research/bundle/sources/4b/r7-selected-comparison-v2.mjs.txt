/** CPU replay of available paired evidence; missing post diagnostics stay explicitly missing. */
import fs from'node:fs';import path from'node:path';import assert from'node:assert/strict';import{fileURLToPath}from'node:url';import{exposure}from'./r7-selected-baseline-v1.mjs';import{fileHash}from'./precision-diagnostic.mjs';import{summarize}from'./r6-score.mjs';import{paired}from'./r3-score.mjs';
const self=fileURLToPath(import.meta.url),read=p=>JSON.parse(fs.readFileSync(p));
export function compareAvailable(baseline,out){
 assert([baseline,out].every(path.isAbsolute));assert(!fs.existsSync(out),'REFUSE_OVERWRITE');const policy=read(path.join(baseline,'POLICY.json')),root=policy.root,summary=read(path.join(baseline,'summary.json')),e=exposure(root),postPath=path.join(root,'post-v1/state.json'),post=read(postPath);assert.equal(read(path.join(root,'run-state.json')).status,'complete-research-only');assert(['failed','complete-research-evidence'].includes(post.status),'POST_NOT_TERMINAL');for(const p of policy.pins)assert.equal(fileHash(p.path),p.sha256,'BASELINE_PIN_CHANGED');assert.equal(summary.selectedAdapterSha256,e.adapterSha256);
 const references={r3:read(path.join(e.parent,'native-reference.json')),selected:read(path.join(root,'native-reference.json'))},reports=[],missing=[],pins=[self,postPath,path.join(baseline,'summary.json'),path.join(baseline,'POLICY.json')].map(p=>({path:p,sha256:fileHash(p)}));
 for(const g of policy.groups){const cases=read(g.file).filter(c=>!g.split||c.split===g.split);assert.equal(cases.length,g.count);const classicalPath=path.join(baseline,g.name+'-scores.json'),classical=read(classicalPath),models={},comparisons=[];pins.push({path:classicalPath,sha256:fileHash(classicalPath)});
  for(const name of['base','r3','selected']){const file=g.name==='core-dev'?path.join(root,name==='selected'?(e.selectedStep?'dev-step-'+e.selectedStep:'r3-dev')+'.json':name+'-dev.json'):g.name==='core-test'?path.join(root,name+'-test.json'):path.join(root,'post-v1',g.name+'-'+name+'.json');
   if(!fs.existsSync(file)){assert(!g.name.startsWith('core-'),'MISSING_COMPLETE_CORE');missing.push({group:g.name,arm:name,path:file,reason:post.error??'NO_RAW_RECEIPT'});continue;}
   const raw=read(file);if(!raw.complete){assert(!g.name.startsWith('core-'));missing.push({group:g.name,arm:name,path:file,reason:'INCOMPLETE_RAW'});continue;}assert.equal(raw.metadata.modelPath,references.r3.base);if(name==='base')assert.equal(raw.metadata.adapter,null);else assert.equal(fileHash(path.join(raw.metadata.adapter,'adapters.safetensors')),references[name].adapterSha256,'MODEL_IDENTITY');const s=summarize(cases,raw);pins.push({path:file,sha256:fileHash(file)});models[name]={total:s.total,passed:s.passed,unsafe:s.unsafe,tasks:s.tasks,rawPath:file,meanGenerationSeconds:raw.results.reduce((n,r)=>n+r.seconds,0)/raw.results.length};
   for(const[b,bs]of Object.entries(classical)){const ids=new Set(bs.rows.map(r=>r.id));comparisons.push({before:b,after:name,byTask:paired(bs,{...s,rows:s.rows.filter(r=>ids.has(r.id))})});}
  }reports.push({name:g.name,classical:Object.fromEntries(Object.entries(classical).map(([n,s])=>[n,{total:s.total,passed:s.passed,unsafe:s.unsafe,tasks:s.tasks}])),models,comparisons});
 }
 const result={createdAt:new Date().toISOString(),status:missing.length?'partial-research-evidence':'complete-research-evidence',postStatus:post.status,postError:post.error??null,baseline,selectedStep:e.selectedStep,selectedAdapterSha256:e.adapterSha256,selectedPrefixRows:e.selectedR7PrefixRows,rawTrainRows:summary.rawTrainRows,uniqueTrainRows:summary.uniqueTrainRows,reports,missing,pins,sameSupervisedExamplesAvailable:true,equalPretrainingOrCompute:false,checkpointReselected:false,algorithmChanged:false,selectionByTest:false,releaseQualified:false};fs.mkdirSync(out);fs.writeFileSync(path.join(out,'summary.json'),JSON.stringify(result,null,2)+'\n',{flag:'wx'});
 const lines=['# 所選R7實際訓練暴露：script與模型對照','',`狀態：${result.status}。原post：${post.status}；缺少${missing.length}組原始輸出，不補零、不冒稱通過。`,`R3的434個獨特train＋所選R7前${e.selectedR7PrefixRows}例；原始${summary.rawTrainRows}、去重${summary.uniqueTrainRows}。TF-IDF演算法與dev選參數不變，不按test換政策。`,'','| 題組 | 任務 | 方法 | 正確 | 誤放行 |','| --- | --- | --- | ---: | ---: |'];for(const r of reports)for(const[n,s]of Object.entries({...r.classical,...r.models}))for(const[t,v]of Object.entries(s.tasks))lines.push(`| ${r.name} | ${t} | ${n} | ${v.passed}/${v.total} | ${v.unsafe} |`);lines.push('','相同監督例可用不等於相同預訓練、梯度權重或計算量。此處沿用原R7輸入／wave標籤，不用校正後答案重算舊輸出。沒有VFX非LLM dev，因此不虛構該基線；模型VFX與未完成回歸仍須另外報告。','','## 缺失原始輸出','',...missing.map(m=>`- ${m.group}/${m.arm}：${m.reason}`),'');fs.writeFileSync(path.join(out,'REPORT.md'),lines.join('\n'),{flag:'wx'});return{out,status:result.status,missing:missing.length,groups:reports.length};
}
if(process.argv[1]&&path.resolve(process.argv[1])===self)console.log(JSON.stringify(compareAvailable(...process.argv.slice(2))));
