import test from'node:test';import assert from'node:assert/strict';import{initialChoice,assertFinished,terminalPredecessors}from'./r8-prepare-run-v1.mjs';
const score=(pass,unsafe=false)=>({rows:[{id:'community-source',task:'owner-mechanism',pass,unsafe}],tasks:{'owner-mechanism':{passed:+pass,unsafe:+unsafe}}});
test('R8 warm start cannot replace R3 with a new-community regression',()=>{const r3={adapter:'r3'},r7={adapter:'r7'},past={r3:score(true),selected:score(true)};assert.equal(initialChoice(r3,r7,{r3:score(true),'r7-selected':score(true)},past).reference,r7);assert.equal(initialChoice(r3,r7,{r3:score(true),'r7-selected':score(false)},past).reference,r3);assert.equal(initialChoice(r3,r7,{r3:score(false),'r7-selected':score(false,true)},past).reference,r3);});
test('known core regression cannot be hidden by a good new-community dev',()=>{const r3={adapter:'r3'},r7={adapter:'r7'},choice=initialChoice(r3,r7,{r3:score(true),'r7-selected':score(true)},{r3:score(true),selected:score(false)});assert.equal(choice.reference,r3);assert.equal(choice.warmStartUsesPastExposedRegression,true);assert.equal(choice.newCommunityTestUsed,false);assert.equal(choice.knownRegressionPreservation.pass,false);});
test('R8 requires successful parent statuses AND terminal process handles',()=>{assert(assertFinished([{status:'done'}],['done'],[false]));assert.throws(()=>assertFinished([{status:'done'}],['done'],[true]),/STILL_LIVE/);assert.throws(()=>assertFinished([{status:'failed'}],['done'],[false]),/INCOMPLETE/);assert.throws(()=>assertFinished([{status:'running'}],['done'],[false]),/INCOMPLETE/);});
test('failed supplemental diagnostics remain failed; only completed core and dead predecessors permit independent R3 continuation',()=>{const expected=Array(5).fill('done'),dead=Array(5).fill(false),states=expected.map(status=>({status}));states[1]={status:'failed',error:'GPU_CUTOFF'};assert.deepEqual(terminalPredecessors(states,expected,dead),['done','failed','done','done','done']);assert.throws(()=>terminalPredecessors(states,expected,[false,true,false,false,false]),/STILL_LIVE/);states[0]={status:'failed',error:'core failed'};assert.throws(()=>terminalPredecessors(states,expected,dead),/NOT_TERMINAL/);});
