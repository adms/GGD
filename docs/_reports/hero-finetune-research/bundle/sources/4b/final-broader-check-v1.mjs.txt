/** Same-input wider regressions on fresh base/R3/native/fused/8bit. No selection or training. */
import fs from'node:fs';import path from'node:path';import assert from'node:assert/strict';import{spawn}from'node:child_process';import{fileURLToPath}from'node:url';import{verify}from'./final-regression-data-v1.mjs';import{fileHash}from'./precision-diagnostic.mjs';import{comparePrecisions,render as renderPrecision}from'./final-precision-score-v1.mjs';import{compare,render}from'./r6-score.mjs';import{validateDiagnostic}from'./r3-fidelity-diagnostic.mjs';
const self=fileURLToPath(import.meta.url),dir=path.dirname(self),read=p=>JSON.parse(fs.readFileSync(p));
export function sliceArms(arms,ids){const wanted=new Set(ids);return arms.map(a=>({...a,raw:{...a.raw,results:a.raw.results.filter(r=>wanted.has(r.id))}}));}
export function scoreGroups(cases,arms,groups){
 assert.deepEqual(arms.map(a=>a.name),['base','r3','native','fused-bf16','mlx-8bit']);return groups.map(g=>{const ids=new Set(g.ids),cs=cases.filter(c=>ids.has(c.id)),subset=sliceArms(arms,g.ids);assert.equal(cs.length,g.count);return{name:g.name,baseComparison:compare(cs,subset.slice(0,3)),precisionComparison:comparePrecisions(cs,subset.slice(2))};});
}

 }catch(e){state.status='failed';state.error=String(e);put('state.json',state);console.error(e);process.exitCode=1;}

}
