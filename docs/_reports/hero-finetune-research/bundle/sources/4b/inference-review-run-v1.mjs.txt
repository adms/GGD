/** Inference-only, chunk-paired replay. No training, checkpoint/precision choice or output repair. */
import fs from'node:fs';import path from'node:path';import assert from'node:assert/strict';import{spawn}from'node:child_process';import{fileURLToPath}from'node:url';import{fileHash}from'./precision-diagnostic.mjs';import{compare}from'./r6-score.mjs';
const self=fileURLToPath(import.meta.url),dir=path.dirname(self),read=p=>JSON.parse(fs.readFileSync(p));
async function main(data,python,runtime,out){
 assert([data,python,runtime,out].every(path.isAbsolute));assert(!fs.existsSync(out),'REFUSE_RESTART');const manifest=read(path.join(data,'manifest.json')),cases=read(path.join(data,'cases.private.json')),budgets=read(path.join(data,'token-budget.json'));assert.equal(manifest.trainingCallsAllowed,false);assert.equal(manifest.checkpointSelectionAllowed,false);assert.equal(cases.length,manifest.uniqueRequests);assert.equal(budgets.count,cases.length);assert(budgets.allFit&&budgets.cpuOnly);assert.equal(budgets.requestsSha256,fileHash(path.join(data,'cases.private.json')));for(const p of manifest.pins)assert.equal(fileHash(p.path),p.sha256,'DATA_PIN_CHANGED');
 const refs={r3:read(path.join(manifest.outputs,'forge-mechanism-priority-r3-20260906/native-reference.json')),r7:read(path.join(manifest.outputs,'forge-low-lr-r7-v2-20260906/native-reference.json'))},base=refs.r3.base;assert.equal(refs.r7.base,base);for(const r of Object.values(refs))assert.equal(fileHash(path.join(r.adapter,'adapters.safetensors')),r.adapterSha256);const disabled=path.join(manifest.outputs,'forge-source-rehearsal-r8-v1-20260906/TRAINING_DISABLED.json');assert.equal(read(disabled).trainingCallsAllowed,false);
 const lock=path.join(runtime,'gpu.lock');assert(!fs.existsSync(lock),'GPU_BUSY');const cutoff=Date.parse('2026-09-06T20:06:10Z');assert(Date.now()<cutoff-300000,'INFERENCE_START_CUTOFF');fs.mkdirSync(out);const put=(n,v)=>{const p=path.join(out,n);fs.writeFileSync(p+'.tmp',JSON.stringify(v,null,2)+'\n');fs.renameSync(p+'.tmp',p);};
 const policy={schema:'ggd-inference-only-review@1',deadline:'2026-09-06T20:36:10Z',reportReserveSeconds:1800,maxMemoryGiB:12,maxSequenceTokens:4096,maxOutputTokens:256,cloudGpu:false,cloudFallback:false,cloudGpuSpendLimit:0,externalTeacher:false,publish:false,activateInEditor:false,thinking:false,seed:20260906,temperature:0,topP:.8,topK:20,presencePenalty:0,trainingCallsAllowed:false,checkpointSelectionAllowed:false,releaseQualified:false};put('policy.json',policy);
 const baseReceiptPath=path.join(path.dirname(base),'NATIVE_REFERENCE.json'),baseReceipt=read(baseReceiptPath),baseFiles=baseReceipt.files.filter(f=>f.name.startsWith('base/'));assert.equal(baseFiles.length,13);const sourcePaths=[self,...['worker.py','r6-score.mjs','r3-score.mjs','classification-score.mjs','composition-client.mjs'].map(n=>path.join(dir,n)),...['manifest.json','cases.private.json','mapping.private.json','token-budget.json'].map(n=>path.join(data,n)),baseReceiptPath,disabled,path.join(out,'policy.json'),...Object.values(refs).flatMap(r=>['adapters.safetensors','adapter_config.json'].map(n=>path.join(r.adapter,n)))],pins=sourcePaths.map(p=>({path:p,sha256:fileHash(p)}));
 put('PREPARATION.json',{preparedAt:new Date().toISOString(),data,manifestSha256:fileHash(path.join(data,'manifest.json')),pins,modelReferences:refs,models:['base','r3','r7'],uniqueRequests:cases.length,chunks:manifest.chunks.length,plannedModelCalls:cases.length*3,cutoff:new Date(cutoff).toISOString(),trained:false,selectionAllowed:false,scope:'Fresh inference across all selected reviewed historical versions; old capability contexts stay labelled historical. Quarantined/obsolete unmatched corpora remain inventoried, not newly admitted. No test-driven changes.'});
 fs.writeFileSync(lock,JSON.stringify({pid:process.pid,kind:'inference-only-dataset-review',root:out}),{flag:'wx'});const state={status:'running',pid:process.pid,startedAt:new Date().toISOString(),stages:[],pairedChunks:[],trainingCalls:0};put('state.json',state);let child=null,reason=null,killTimer;
 function guard(){assert(!reason,reason);assert(Date.now()<cutoff,'GPU_CUTOFF');for(const r of[out,data,path.dirname(data)])assert(!fs.existsSync(path.join(r,'CANCEL')),'CANCEL');assert.equal(read(disabled).trainingCallsAllowed,false);}
 function verifyPins(){guard();for(const p of pins)assert.equal(fileHash(p.path),p.sha256,'PIN_CHANGED:'+p.path);}
 function modelHashes(){for(const f of baseFiles)assert.equal(fileHash(path.join(path.dirname(base),f.name)),f.sha256,'BASE_CHANGED');for(const r of Object.values(refs))assert.equal(fileHash(path.join(r.adapter,'adapters.safetensors')),r.adapterSha256);return{checkedAt:new Date().toISOString(),baseFiles:baseFiles.length,adapterSha256:Object.fromEntries(Object.entries(refs).map(([n,r])=>[n,r.adapterSha256]))};}
 function stop(why){reason??=why;if(child?.pid){try{process.kill(-child.pid,'SIGTERM');}catch{}killTimer=setTimeout(()=>{try{process.kill(-child.pid,'SIGKILL');}catch{}},3000);}}
 process.on('SIGTERM',()=>stop('SIGNAL'));process.on('SIGINT',()=>stop('SIGNAL'));
 async function evaluate(chunk,name){verifyPins();const request=path.join(data,chunk.request);assert.equal(fileHash(request),chunk.requestsSha256);const stage=chunk.name+'-'+name,output=path.join(out,stage+'.json'),adapter=name==='base'?null:refs[name].adapter,args=[path.join(dir,'worker.py'),'eval','--model',base,'--policy',path.join(out,'policy.json'),'--data',request,'--output',output];if(adapter)args.push('--adapter',adapter);state.stage=stage;put('state.json',state);const log=fs.openSync(path.join(out,stage+'.log'),'wx');child=spawn(python,args,{stdio:['ignore',log,log],detached:true});fs.closeSync(log);put('worker-lease.json',{pid:child.pid,parent:process.pid,stage});const timer=setInterval(()=>{try{guard();}catch(e){stop(String(e));}},2000);let code;try{code=await new Promise((res,rej)=>{child.once('exit',res);child.once('error',rej);});}finally{clearInterval(timer);clearTimeout(killTimer);child=null;put('worker-lease.json',{pid:null,stage});}assert(!reason&&code===0,reason??'WORKER_EXIT:'+code);verifyPins();assert.equal(fileHash(request),chunk.requestsSha256);const raw=read(output);assert(raw.complete&&raw.results.length===chunk.count);assert.equal(raw.metadata.modelPath,base);assert.equal(raw.metadata.adapter,adapter);assert(raw.metadata.peakMetalBytes<=12*1024**3);state.stages.push({stage,completedAt:new Date().toISOString(),rows:chunk.count});put('state.json',state);console.log(JSON.stringify(state.stages.at(-1)));return{name,raw};}
 try{put('models-before.json',modelHashes());for(const chunk of manifest.chunks){const wanted=new Set(chunk.ids),rows=cases.filter(c=>wanted.has(c.id));assert.deepEqual(rows.map(c=>c.id),chunk.ids);const arms=[];for(const name of['base','r3','r7'])arms.push(await evaluate(chunk,name));const report=compare(rows,arms);put(chunk.name+'-comparison.json',report);state.pairedChunks.push({name:chunk.name,rows:chunk.count,completedAt:new Date().toISOString(),scores:Object.fromEntries(Object.entries(report.scores).map(([n,s])=>[n,{passed:s.passed,total:s.total,unsafe:s.unsafe}]))});put('state.json',state);put('partial-summary.json',{state,plannedRows:cases.length,pairedRows:state.pairedChunks.reduce((n,c)=>n+c.rows,0),trainingCalls:0,releaseQualified:false});}
  put('models-after.json',modelHashes());state.status='complete-research-evidence';state.completedAt=new Date().toISOString();put('state.json',state);
 }catch(e){state.status='failed';state.error=String(e);state.endedAt=new Date().toISOString();put('state.json',state);console.error(e);process.exitCode=1;}
 finally{clearTimeout(killTimer);if(fs.existsSync(lock)&&read(lock).pid===process.pid)fs.unlinkSync(lock);put('worker-lease.json',{pid:null,reason,endedAt:new Date().toISOString()});put('summary.json',{status:state.status,error:state.error??null,plannedRequests:cases.length,plannedModelCalls:cases.length*3,completedModelCalls:state.stages.reduce((n,s)=>n+s.rows,0),pairedRequests:state.pairedChunks.reduce((n,s)=>n+s.rows,0),pairedChunks:state.pairedChunks,missingChunks:manifest.chunks.filter(c=>!state.pairedChunks.some(s=>s.name===c.name)).map(c=>({name:c.name,rows:c.count})),trainingCalls:0,selectionAllowed:false,releaseQualified:false});}
}
if(process.argv[1]&&path.resolve(process.argv[1])===self)await main(...process.argv.slice(2));
