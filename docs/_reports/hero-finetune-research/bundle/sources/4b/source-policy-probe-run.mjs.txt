/** Dev-first, error-informed prompt probe; never fine-tunes or reselects R4. */
import fs from 'node:fs';import path from 'node:path';import assert from 'node:assert/strict';import {spawn} from 'node:child_process';import {fileURLToPath} from 'node:url';import {createHash} from 'node:crypto';
import {digest} from './dataset.mjs';import {compareArms,renderReport} from './r3-error-report.mjs';import {eligible} from './r3-score.mjs';
const [root,python,runtime,phase]=process.argv.slice(2);for(const p of [root,python,runtime])assert(p&&path.isAbsolute(p));assert(['dev','regression'].includes(phase));
const dir=path.dirname(fileURLToPath(import.meta.url)),read=p=>JSON.parse(fs.readFileSync(p)),hash=p=>createHash('sha256').update(fs.readFileSync(p)).digest('hex'),manifest=read(path.join(root,'manifest.json')),r4=manifest.sourceRoot,r3=read(path.join(r4,'dataset-manifest.json')).parent;
assert.equal(manifest.schema,'ggd-source-policy-probe@1');assert.equal(read(path.join(r4,'post-diagnostics-v2/state.json')).status,'complete-research-evidence','WAIT_FOR_POST');assert.equal(read(path.join(r4,'supplement-v1/state.json')).status,'complete-research-evidence','WAIT_FOR_SUPPLEMENT');
const all=read(path.join(root,'cases.private.json'));assert.equal(digest(all),manifest.casesSha256);const split=phase==='dev'?'dev':'test',cases=all.filter(c=>c.split===split),requests=path.join(root,split+'-requests.json');assert.deepEqual(read(requests),cases.map(({id,messages,requestDigest})=>({id,messages,requestDigest})));
if(phase==='regression')assert.equal(read(path.join(root,'dev/summary.json')).devGate,true,'DEV_NOT_ELIGIBLE');
const policyPath=path.join(r4,'experiment-policy.json'),policy=read(policyPath),cutoff=Date.parse(policy.deadline)-policy.reportReserveSeconds*1000,native=read(path.join(r3,'native-reference.json'));assert.equal(hash(path.join(native.adapter,'adapters.safetensors')),native.adapterSha256);
for(const k of ['cloudGpu','cloudFallback','externalTeacher','publish','activateInEditor','thinking'])assert.equal(policy[k],false);assert.equal(policy.cloudGpuSpendLimit,0);
const lock=path.join(runtime,'gpu.lock'),out=path.join(root,phase);assert(!fs.existsSync(out),'REFUSE_RESTART');assert(!fs.existsSync(lock),'GPU_BUSY');assert(Date.now()+300000<cutoff,'INSUFFICIENT_RESERVE');
const files=[fileURLToPath(import.meta.url),policyPath,requests,path.join(root,'cases.private.json'),path.join(root,'manifest.json'),...['worker.py','source-policy-probe-data.mjs','r3-error-report.mjs','r3-score.mjs','classification-score.mjs','dataset.mjs'].map(n=>path.join(dir,n))],pins=files.map(p=>({path:p,sha256:hash(p)}));
fs.mkdirSync(out);const put=(n,x)=>{const p=path.join(out,n);fs.writeFileSync(p+'.tmp',JSON.stringify(x,null,2)+'\n');fs.renameSync(p+'.tmp',p);};put('pins.json',{pins,base:native.base,adapter:native.adapter,adapterSha256:native.adapterSha256,policyVersion:manifest.clarificationSha256,weightUpdate:false});
let child=null,reason=null,killTimer,owned=false;const state={status:'running',phase,pid:process.pid,startedAt:new Date().toISOString(),stages:[]};put('state.json',state);
function stop(why){if(reason)return;reason=why;if(child?.pid){try{process.kill(-child.pid,'SIGTERM');}catch{}killTimer=setTimeout(()=>{try{process.kill(-child.pid,'SIGKILL');}catch{}},3000);}}
function guard(){assert(!reason,reason);assert(Date.now()<cutoff,'DEADLINE');assert(!fs.existsSync(path.join(root,'CANCEL')),'CANCEL');for(const p of pins)assert.equal(hash(p.path),p.sha256,'PIN_CHANGED:'+p.path);}
process.on('SIGTERM',()=>stop('SIGNAL'));process.on('SIGINT',()=>stop('SIGNAL'));
try{
 fs.writeFileSync(lock,JSON.stringify({pid:process.pid,kind:'source-policy-probe',root:out}),{flag:'wx'});owned=true;const arms=[];
 for(const [name,adapter] of [['base',null],['r3',native.adapter]]){
  guard();state.stage=name;put('state.json',state);const output=path.join(out,name+'.json'),args=[path.join(dir,'worker.py'),'eval','--model',native.base,'--policy',policyPath,'--data',requests,'--output',output];if(adapter)args.push('--adapter',adapter);
  const log=fs.openSync(path.join(out,name+'.log'),'wx');child=spawn(python,args,{stdio:['ignore',log,log],detached:true});fs.closeSync(log);put('worker-lease.json',{pid:child.pid,parent:process.pid});
  const timer=setInterval(()=>{try{guard();}catch(e){stop(String(e));}},2000);let code;try{code=await new Promise((resolve,reject)=>{child.once('error',reject);child.once('exit',resolve);});}finally{clearInterval(timer);clearTimeout(killTimer);child=null;put('worker-lease.json',{pid:null});}
  assert(!reason&&code===0,reason??'WORKER_EXIT:'+name+':'+code);guard();const raw=read(output);assert.equal(raw.metadata.modelPath,native.base);assert.equal(raw.metadata.adapter,adapter);arms.push({name,raw});state.stages.push({name,completedAt:new Date().toISOString()});put('state.json',state);console.log(JSON.stringify(state.stages.at(-1)));
 }
 const report=compareArms(cases,arms);put('report.json',report);fs.writeFileSync(path.join(out,'REPORT.md'),renderReport(report),{flag:'wx'});
 const summary={phase,scores:Object.fromEntries(Object.entries(report.scores).map(([n,s])=>[n,{total:s.total,passed:s.passed,unsafe:s.unsafe,tasks:s.tasks}])),devGate:phase==='dev'?eligible(report.scores.r3):null,weightUpdate:false,checkpointReselected:false,releaseQualified:false,scope:'Prompt-policy intervention with equally prompted base/R3 arms. All source questions already exposed in R4; not new generalization or additional fine-tuning evidence.'};put('summary.json',summary);state.status='complete-research-evidence';state.completedAt=new Date().toISOString();put('state.json',state);console.log(JSON.stringify(summary));
}catch(e){state.status='failed';state.error=String(e);put('state.json',state);console.error(e);process.exitCode=1;}
finally{if(owned&&fs.existsSync(lock)&&read(lock).pid===process.pid)fs.unlinkSync(lock);put('worker-lease.json',{pid:null,reason,endedAt:new Date().toISOString()});}
