import test from'node:test';import assert from'node:assert/strict';import{merge}from'./inference-review-data-v1.mjs';
test('deduplication retains every split and exposes contradictory labels without silently choosing a gold answer',()=>{const c={id:'a',task:'hero-source',split:'train',messages:[{role:'system',content:'source'},{role:'user',content:'same'}],requestDigest:'x',target:{verdict:'supported'},acceptedTargets:[{verdict:'supported'}]};const rows=merge([{name:'old',cases:[c]},{name:'new',cases:[{...c,id:'b',split:'test'}]}]);assert.equal(rows.length,1);assert.equal(rows[0].members.length,2);assert.equal(rows[0].conflictingLabels,false);const conflict=merge([{name:'old',cases:[c]},{name:'new',cases:[{...c,acceptedTargets:[{verdict:'contradicted'}]}]}]);assert.equal(conflict[0].conflictingLabels,true);assert.equal(conflict[0].members[0].expected[0].verdict,'supported');assert.equal(conflict[0].members[1].expected[0].verdict,'contradicted');});
