/** Evidence-only collection; no training, checkpoint reselection or activation. */
import fs from 'node:fs';import path from 'node:path';import assert from 'node:assert/strict';import {fileURLToPath} from 'node:url';
import {fileHash} from './precision-diagnostic.mjs';import {compare,summarize,preservation,rank,eligible} from './r6-score.mjs';import {prepare,validate} from './research-inference-v2.mjs';import {isDeepStrictEqual as equal} from 'node:util';
const read=p=>JSON.parse(fs.readFileSync(p));
export const canonical=x=>JSON.parse(JSON.stringify(x));
const measuredTaskPass=t=>!!t&&t.pct>=95&&t.unsafe===0&&t.schema===t.total;
export function collect(root){
 const states=['run-state.json','post-v1/state.json','manual-entry-v3/state.json'].map(n=>read(path.join(root,n)));assert.equal(states[0].status,'complete-research-only','INCOMPLETE_CORE');for(const s of states.slice(1))assert.equal(s.status,'complete-research-evidence','INCOMPLETE_POST_OR_ENTRY');
 const policy=read(path.join(root,'experiment-policy.json'));assert.equal(policy.presencePenalty,0);for(const k of ['cloudGpu','publish','activateInEditor','thinking'])assert.equal(policy[k],false);
 const corePins=read(path.join(root,'run-pins.json')).pins,postPins=read(path.join(root,'post-v1/pins.json')).pins,selectedPins=read(path.join(root,'post-v1/selected-pins.json')).pins,entryPrep=read(path.join(root,'manual-entry-v3/preparation.json'));for(const p of [...corePins,...postPins,...selectedPins,...entryPrep.pins])assert.equal(fileHash(p.path),p.sha256,'PIN_CHANGED:'+p.path);
 const manifest=read(path.join(root,'dataset-manifest.json')),cases=read(path.join(root,'cases.private.json')),dev=cases.filter(c=>c.split==='dev'),test=cases.filter(c=>c.split==='test'),control=read(path.join(root,'r3-dev.json')),controlScore=summarize(dev,control),selection=read(path.join(root,'selection.json'));
 assert.equal(selection.testUsed,false);const replay=selection.allCandidates.map(c=>{const raw=c.step===0?control:read(path.join(root,'dev-step-'+c.step+'.json')),score=summarize(dev,raw);assert.deepEqual(score,c.score);assert.deepEqual(preservation(controlScore,score),c.preservation);return{...c,score};}).sort(rank);assert.deepEqual(replay[0],selection.selected,'SELECTION_NOT_REPLAYABLE');
 const native=read(path.join(root,'native-reference.json')),training=read(path.join(root,'training-run.json'));assert.equal(native.selectedStep,selection.selected.step);assert.equal(fileHash(path.join(native.adapter,'adapters.safetensors')),native.adapterSha256);assert.equal(native.adapterSha256,selection.selected.adapterSha256);assert.equal(native.decoding.presencePenalty,0);
 const devReport=compare(dev,[{name:'base',raw:read(path.join(root,'base-dev.json'))},{name:'r3',raw:control},{name:'selected',raw:selection.selected.step?read(path.join(root,'dev-step-'+selection.selected.step+'.json')):control}]);assert.deepEqual(canonical(devReport),read(path.join(root,'dev-report.json')));
 const testReport=compare(test,['base','r3','selected'].map(name=>({name,raw:read(path.join(root,name+'-test.json'))})));assert.deepEqual(canonical(testReport),read(path.join(root,'test-report.json')));
 const summary=read(path.join(root,'post-v1/summary.json')),r3=manifest.parent,r4=manifest.previousRegression,ext=manifest.extension,outputs=path.dirname(root),datasets=[
  ['new-source-63',path.join(root,'fresh-source-v1'),'cases.private.json'],['previous-source-72',path.join(outputs,'forge-contrastive-r5-20260906/fresh-source-diagnostic-v1'),'cases.private.json'],['current-main-v4',path.join(r3,'main-diagnostic-v4'),'cases.private.json'],['hero-140',path.join(r3,'main-hero-diagnostic-v1'),'cases.private.json'],['fidelity-114',path.join(r3,'fidelity-diagnostic-v2'),'cases.private.json'],['vfx-118',path.join(r4,'post-diagnostics-v2'),'vfx-cases.private.json'],['composition-regression-20',path.join(ext,'composition-diagnostic-v2'),'cases.private.json'],['source-clarification-15',path.join(ext,'source-label-clarification-v1'),'cases.private.json'],
 ];
 const post={};for(const [name,where,file] of datasets){const c=read(path.join(where,file)),rep=compare(c,['base','r3','selected'].map(arm=>({name:arm,raw:read(path.join(root,'post-v1',name+'-'+arm+'.json'))})));assert.deepEqual(canonical(rep),read(path.join(root,'post-v1',name+'-report.json')));post[name]=Object.fromEntries(Object.entries(rep.scores).map(([n,s])=>[n,{total:s.total,passed:s.passed,unsafe:s.unsafe,tasks:s.tasks}]));assert.deepEqual(post[name],summary.reports.find(r=>r.name===name).scores);}
 const entry=read(path.join(root,'manual-entry-v3/summary.json')),expected=read(path.join(root,'manual-entry-v3/EXPECTED.private.json'));assert.equal(entry.results.length,10);
 for(const r of entry.results){const spec=read(path.join(root,'manual-entry-v3',r.id+'.json')),prepared=prepare(spec),raw=read(path.join(root,'manual-entry-v3',r.id+'-result/raw.json')),checked=validate(spec,prepared,raw);assert.equal(checked.contractPass,r.contractPass);assert.deepEqual(checked.values,r.values);assert.equal(r.semanticMatch,checked.contractPass&&expected.cases.find(c=>c.id===r.id).acceptedValues.some(v=>equal(v,r.values.map(x=>x.value))));assert.equal(r.adapterSha256,native.adapterSha256);}
 assert.equal(entry.semanticPassed,entry.results.filter(r=>r.semanticMatch).length);assert.equal(entry.allSemanticsPass,entry.results.every(r=>r.semanticMatch));
 const gates={dev:eligible(selection.selected.score),coreTest:eligible(testReport.scores.selected),newSource63:measuredTaskPass(post['new-source-63'].selected.tasks['owner-mechanism']),previousSource72:measuredTaskPass(post['previous-source-72'].selected.tasks['owner-mechanism']),currentMain40:measuredTaskPass(post['current-main-v4'].selected.tasks['mechanism-template']),hero140:measuredTaskPass(post['hero-140'].selected.tasks['hero-source']),fidelity114:measuredTaskPass(post['fidelity-114'].selected.tasks['owner-mechanism']),composition20:measuredTaskPass(post['composition-regression-20'].selected.tasks['mechanism-stack']),clarification15:measuredTaskPass(post['source-clarification-15'].selected.tasks['owner-mechanism']),publicEntry:entry.allSemanticsPass};
 const deltas=Object.fromEntries(Object.keys(testReport.scores.selected.tasks).map(task=>{const a=testReport.scores.base.tasks[task],b=testReport.scores.r3.tasks[task],c=testReport.scores.selected.tasks[task];return[task,{vsBaseCorrect:c.passed-a.passed,vsBasePercentagePoints:c.pct-a.pct,vsR3Correct:c.passed-b.passed,vsR3PercentagePoints:c.pct-b.pct,vsBaseUnsafe:c.unsafe-a.unsafe,vsR3Unsafe:c.unsafe-b.unsafe}];}));
 const baseReceipt=read(path.join(path.dirname(native.base),'NATIVE_REFERENCE.json'));for(const f of baseReceipt.files.filter(f=>f.name.startsWith('base/'))){const p=path.join(path.dirname(native.base),f.name);assert.equal(fs.statSync(p).size,f.bytes);assert.equal(fileHash(p),f.sha256,'BASE_BYTES_CHANGED');}
 return{collectedAt:new Date().toISOString(),root,selectedStep:native.selectedStep,newAdapterSelected:native.selectedStep!==0,adapterSha256:native.adapterSha256,modelReference:path.join(root,'native-reference.json'),trainingComplete:training.trainingComplete,training:{prepared:training.preparedExamples,processed:training.processedExamples,committed:training.committedExamples,seconds:training.totalSeconds,peakMetalGiB:training.peakMetalBytes/1024**3},devScores:Object.fromEntries(Object.entries(read(path.join(root,'dev-report.json')).scores).map(([n,s])=>[n,{total:s.total,passed:s.passed,unsafe:s.unsafe,tasks:s.tasks}])),testScores:Object.fromEntries(Object.entries(testReport.scores).map(([n,s])=>[n,{total:s.total,passed:s.passed,unsafe:s.unsafe,tasks:s.tasks}])),deltas,post,entry:{total:10,passed:entry.semanticPassed,allContractsPass:entry.allContractsPass,allSemanticsPass:entry.allSemanticsPass},measuredGates:gates,allMeasuredHighPriorityGatesPass:Object.values(gates).every(Boolean),physical16GBMachineTested:false,fusedOrQuantizedQualityTested:false,releaseQualified:false,activation:false,baseBytesVerified:true,oldScoresRewritten:false,decodingPresencePenalty:0,limitations:['158 dev selects only; original162 core test is exposed regression, appended24 stack is new wording on known plans.','New63 repository skill texts are new relative to audited files, not latest Owner, new franchise families or independent human Gold.','High-priority measured gates use >=95 percent, zero unsafe and full contract per group. VFX is reported separately and cannot offset source/mechanism failures.','Even all measured gates passing does not establish all heroes/mechanisms/unlisted combinations, engine E2E, 16GB hardware or quantized quality.','No automatic activation or publication; joint training and decode changes are separated by rerunning all controls with zero penalty.']};
}
export function render(result){
 const lines=['# R6-v2 實際實驗結果','',result.newAdapterSelected?`選中step${result.selectedStep}，但仍須逐項看門檻，不因選中就宣布可用。`:'守衛選模保留原R3（step0）；本輪沒有採用新微調權重，不得把R3原本的收益說成R6改善。','',`訓練${result.trainingComplete?'完整一輪':'僅部分完成'}：${result.training.committed}/${result.training.prepared}筆完整更新，${(result.training.seconds/60).toFixed(1)}分鐘。解碼全模型presence penalty=0；歷史1.5分數不混用。`,'','## 核心同條件測試','','| 任務 | base | R3 | 選定 | 選定錯誤放行 | 對base百分點 | 對R3百分點 |','| --- | ---: | ---: | ---: | ---: | ---: | ---: |'];
 for(const [task,t] of Object.entries(result.testScores.selected.tasks)){const a=result.testScores.base.tasks[task],b=result.testScores.r3.tasks[task],d=result.deltas[task];lines.push(`| ${task} | ${a.passed}/${a.total} | ${b.passed}/${b.total} | ${t.passed}/${t.total} | ${t.unsafe} | ${d.vsBasePercentagePoints.toFixed(2)} | ${d.vsR3PercentagePoints.toFixed(2)} |`);}
 lines.push('','## 分開呈現的補測','','每格為「正確/題數；錯誤放行」。不能把來源、模板、多卡和VFX加總平均掩蓋退步。','','| 題組 | base | R3 | 選定 |','| --- | --- | --- | --- |');for(const [name,p] of Object.entries(result.post))lines.push(`| ${name} | ${['base','r3','selected'].map(n=>`${p[n].passed}/${p[n].total}；${p[n].unsafe}`).join(' | ')} |`);
 lines.push('',`真實公開入口：${result.entry.passed}/${result.entry.total}次語意符合預期；12GiB分配限制，不是16GB實機。`,'','## 門檻與處置','',...Object.entries(result.measuredGates).map(([k,v])=>`- ${k}: ${v?'PASS':'FAIL'}`),'',`所有已測高優先門檻：${result.allMeasuredHighPriorityGatesPass?'PASS（仍限已測範圍）':'FAIL，不得宣稱穩定全用途模型'}`,'','模型位元組及dev選模已重播驗證；原始錯題、來源和模型輸出見core與post的report.json。資料採用／排除見DATA_ADMISSION。未融合量化、未16GB實機、不啟用Editor、不發布。','',...result.limitations.map(s=>'- '+s),'');return lines.join('\n');
}
if(process.argv[1]&&path.resolve(process.argv[1])===fileURLToPath(import.meta.url)){const root=path.resolve(process.argv[2]),out=path.join(root,'final-evidence-v1');assert(!fs.existsSync(out));const result=collect(root);fs.mkdirSync(out);fs.writeFileSync(path.join(out,'summary.json'),JSON.stringify(result,null,2)+'\n',{flag:'wx'});fs.writeFileSync(path.join(out,'REPORT.md'),render(result),{flag:'wx'});console.log(JSON.stringify({out,selectedStep:result.selectedStep,newAdapterSelected:result.newAdapterSelected,measuredGates:result.measuredGates,releaseQualified:false}));}
