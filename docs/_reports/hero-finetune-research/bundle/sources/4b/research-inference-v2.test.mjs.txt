import fs from 'node:fs';import os from 'node:os';import path from 'node:path';import test from 'node:test';import assert from 'node:assert/strict';import {digest} from './dataset.mjs';import {fileHash} from './precision-diagnostic.mjs';import {buildSource,prepare,validate,preflight} from './research-inference-v2.mjs';
const text='「只是一句台詞」\n普通攻擊有10%機率追加傷害。',spec={kind:'source',input:{id:'source-demo',task:'owner-mechanism',source:{id:'fixture-only',name:'CPU fixture',version:'fixture-v1',text,sha256:digest(text)},claim:'普通攻擊有10%機率追加傷害。'}};
test('source version/hash locked, dialogue removed only from mechanics evidence',()=>{const r=buildSource(spec.input);assert.equal(r.sourceLock.version,'fixture-v1');assert(!JSON.parse(r.messages[1].content).source.text.includes('台詞'));assert.throws(()=>buildSource({...spec.input,source:{...spec.input.source,text:text+'drift'}}),/SOURCE_CHANGED/);assert.throws(()=>prepare({...spec,expected:'supported'}));});
test('well-shaped verdict is never semantic approval; extra fields rejected',()=>{const p=prepare(spec),raw={complete:true,metadata:{thinking:false},results:[{...p.requests[0],error:null,value:{verdict:'supported'}}]};const v=validate(spec,p,raw);assert.equal(v.contractPass,true);assert.equal(v.semanticQualified,false);raw.results[0].value.explanation='injected';assert.equal(validate(spec,p,raw).contractPass,false);});
test('GPU busy prevents output creation and no inference; fake bytes only a fixture',()=>{const tmp=fs.mkdtempSync(path.join(os.tmpdir(),'r6-public-fixture-')),adapter=path.join(tmp,'adapter'),base=path.join(tmp,'base');fs.mkdirSync(adapter);fs.mkdirSync(base);fs.writeFileSync(path.join(adapter,'adapters.safetensors'),'not-real-weights');const input=path.join(tmp,'spec.json'),native=path.join(tmp,'native.json');fs.writeFileSync(input,JSON.stringify(spec));fs.writeFileSync(native,JSON.stringify({base,adapter,adapterSha256:fileHash(path.join(adapter,'adapters.safetensors')),decoding:{presencePenalty:0}}));fs.writeFileSync(path.join(tmp,'gpu.lock'),'fixture');assert.throws(()=>preflight(input,native,tmp,path.join(tmp,'output')),/GPU_BUSY/);assert(!fs.existsSync(path.join(tmp,'output')));});
