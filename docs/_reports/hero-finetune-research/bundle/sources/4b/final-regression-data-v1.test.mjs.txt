import fs from'node:fs';import path from'node:path';import test from'node:test';import assert from'node:assert/strict';import{fileURLToPath}from'node:url';import{definitions,build}from'./final-regression-data-v1.mjs';import{correctedDescriptions}from'./r8-data-v1.mjs';
const outputs=path.resolve(path.dirname(fileURLToPath(import.meta.url)),'../../../outputs'),read=p=>JSON.parse(fs.readFileSync(p)),groups=definitions.map(([name,relative,count,prefix=''])=>({name,count,cases:read(path.join(outputs,relative,prefix+'cases.private.json'))})),train=read(path.join(outputs,'forge-source-rehearsal-r8-v1-20260906/cases.private.json')).filter(c=>c.split==='train');
test('all 582 regression cases retained; only evidence-backed terminal label changes',()=>{const original=structuredClone(groups),d=build(groups,train);assert.deepEqual(groups,original);assert.equal(d.labels.length,1);assert.equal(d.labels[0].id,'main-diagnostic-v4-wave-terminal');assert.deepEqual(d.labels[0].revisedTarget,{decision:'accept',templateId:'tpl-line-blast'});const sweep=d.cases.find(c=>c.id==='main-diagnostic-v4-segmented-sweep');assert(sweep.acceptedTargets.some(t=>t.templateId==='tpl-traveling-wave'));assert.equal(JSON.parse(sweep.messages[1].content).catalog.find(t=>t.templateId==='tpl-traveling-wave').description,correctedDescriptions['tpl-traveling-wave']);for(const g of groups)for(const old of g.cases)if(['hero-source','owner-mechanism'].includes(old.task))assert.deepEqual(d.cases.find(c=>c.id===old.id),old);});
test('unexpected terminal wording or missing cases fail before evaluation',()=>{const x=structuredClone(groups),c=x.flatMap(g=>g.cases).find(c=>c.id==='main-diagnostic-v4-wave-terminal'),u=JSON.parse(c.messages[1].content);u.request+='任意獨立節拍。';c.messages[1].content=JSON.stringify(u);assert.throws(()=>build(x,train));const y=structuredClone(groups);y[0].cases.pop();assert.throws(()=>build(y,train));});
