import test from'node:test';import assert from'node:assert/strict';import fs from'node:fs';import path from'node:path';import{fileURLToPath}from'node:url';import{prefixRows,exposure}from'./r7-selected-baseline-v1.mjs';import{fitExamples}from'./nonllm-baseline.mjs';import{fileHash}from'./precision-diagnostic.mjs';
test('exposure takes only the committed selected prefix, never later processed examples',()=>{const rows=['a','b','c','d'].map(id=>({id,split:'train'})),order=['c','a','d','b'];assert.deepEqual(prefixRows(rows,order,2).map(r=>r.id),['c','a']);assert.deepEqual(prefixRows(rows,order,0),[]);assert.throws(()=>prefixRows(rows,order,5));assert.throws(()=>prefixRows(rows,['c','c','a','b'],2));assert.throws(()=>prefixRows(rows.map(c=>c.id==='d'?{...c,split:'test'}:c),order,2));});
test('real R7 selected step48 exposure is 434 inherited plus 192, excluding both withdrawn labels',()=>{const root=path.resolve(path.dirname(fileURLToPath(import.meta.url)),'../../../outputs/forge-low-lr-r7-v2-20260906'),p=path.join(root,'cases.private.json'),before=fileHash(p),e=exposure(root);assert.equal(e.selectedStep,48);assert.equal(e.selectedR7PrefixRows,192);assert.equal(e.train.length,626);assert(e.withdrawnWaveLabelExposure.every(r=>!r.seenBySelectedR7));assert.equal(e.r7WorkerCommitted,380);assert.equal(fileHash(p),before);const m=fitExamples(e.train);assert.equal(m.trainingRows,626);assert(m.uniqueRows<626);assert.equal(fs.existsSync(path.join(root,'CANCEL')),false);});
