/** CPU-only classical comparison using labels the dev-selected R7 weights actually saw. */
import fs from'node:fs';import path from'node:path';import assert from'node:assert/strict';import{fileURLToPath}from'node:url';import{fileHash}from'./precision-diagnostic.mjs';import{digest}from'./dataset.mjs';import{fitExamples,chooseNeighbors,catalogRanking,chooseCatalog,kGrid,thresholdGrid,publicInput}from'./nonllm-baseline.mjs';import{selectDev}from'./nonllm-baseline-run.mjs';import{summarize}from'./r6-score.mjs';import{paired}from'./r3-score.mjs';
const read=p=>JSON.parse(fs.readFileSync(p)),self=fileURLToPath(import.meta.url),dir=path.dirname(self);
export function prefixRows(cases,order,count){assert(Number.isInteger(count)&&count>=0&&count<=order.length);assert.equal(new Set(order).size,order.length);const byId=new Map(cases.filter(c=>c.split==='train').map(c=>[c.id,c]));assert.deepEqual([...byId.keys()].sort(),order.slice().sort());return order.slice(0,count).map(id=>structuredClone(byId.get(id)));}
export function exposure(root){const manifest=read(path.join(root,'dataset-manifest.json')),selection=read(path.join(root,'selection.json')),native=read(path.join(root,'native-reference.json')),policy=read(path.join(root,'experiment-policy.json')),r3=read(path.join(manifest.parent,'native-reference.json')),parentSelection=read(path.join(manifest.parent,'selection.json'));assert.equal(selection.testUsed,false);assert.equal(selection.selected.adapterSha256,native.adapterSha256);assert.equal(fileHash(path.join(native.adapter,'adapters.safetensors')),native.adapterSha256);assert.equal(policy.initialAdapterSha256,r3.adapterSha256);assert.equal(parentSelection.selected.adapterSha256,r3.adapterSha256);assert.equal(parentSelection.selected.epoch,2);
 const cases=read(path.join(root,'cases.private.json')),order=read(path.join(root,'checkpoint-archive/training-order.json')),training=read(path.join(root,'training-run.json')),cp=training.checkpoints.find(c=>c.step===selection.selected.step);let count=0;
 if(selection.selected.step){assert(cp&&cp.weightUpdateComplete);assert.equal(cp.sha256,native.adapterSha256);assert.equal(fileHash(path.join(root,'checkpoint-archive','step-'+cp.step,'adapters.safetensors')),cp.sha256);count=cp.examplesSeen;}else assert.equal(native.adapterSha256,r3.adapterSha256);
 const inherited=read(path.join(manifest.parent,'cases.private.json')).filter(c=>c.split==='train'),prefix=prefixRows(cases,order,count);assert.equal(inherited.length,434);
 return{parent:manifest.parent,selectedStep:selection.selected.step,adapterSha256:native.adapterSha256,inheritedR3Epochs:2,inheritedR3UniqueRows:434,selectedR7PrefixRows:count,r7WorkerProcessed:training.processedExamples,r7WorkerCommitted:training.committedExamples,preparedR7Rows:466,train:[...inherited,...prefix],prefixIds:prefix.map(c=>c.id),withdrawnWaveLabelExposure:['main-candidate-traveling-wave-supported','r6-stack-main-candidate-traveling-wave-supported'].map(id=>({id,position:order.indexOf(id)+1,seenBySelectedR7:prefix.some(c=>c.id===id)})),scope:'Same supervised examples made available to this classical method; not equal pretraining, gradient weighting, capacity or learning algorithm. R3 repeated epochs are not replicated because fitExamples deduplicates identical public-input/target pairs.'};}
function datasets(root,parent){const outputs=path.dirname(root);return[
 {name:'core-dev',file:path.join(root,'cases.private.json'),split:'dev'},
 {name:'core-test',file:path.join(root,'cases.private.json'),split:'test'},
 {name:'source-63-regression',file:path.join(root,'fresh-source-v1/cases.private.json')},
 {name:'previous-source-72',file:path.join(outputs,'forge-contrastive-r5-20260906/fresh-source-diagnostic-v1/cases.private.json')},
 {name:'current-main-v4',file:path.join(parent,'main-diagnostic-v4/cases.private.json')},
 {name:'hero-140',file:path.join(parent,'main-hero-diagnostic-v1/cases.private.json')},
 {name:'fidelity-114',file:path.join(parent,'fidelity-diagnostic-v2/cases.private.json')},
 {name:'composition-regression-20',file:path.join(outputs,'forge-additional-four-hours-20260906/composition-diagnostic-v2/cases.private.json')},
 {name:'source-clarification-15',file:path.join(outputs,'forge-additional-four-hours-20260906/source-label-clarification-v1/cases.private.json')},
 ].map(d=>({...d,cases:read(d.file).filter(c=>!d.split||c.split===d.split)}));}
export function run(root,out){assert([root,out].every(path.isAbsolute));assert(!fs.existsSync(out),'REFUSE_OVERWRITE');const e=exposure(root),groups=datasets(root,e.parent),dev=groups[0].cases,sourceTrain=new Set(e.train.filter(c=>['hero-source','owner-mechanism'].includes(c.task)).map(c=>digest(JSON.parse(c.messages[1].content).source.text))),publicTrain=new Set(e.train.map(c=>digest(publicInput(c.messages))));
 for(const g of groups){g.exactPublicTrainOverlap=g.cases.filter(c=>publicTrain.has(digest(publicInput(c.messages)))).map(c=>c.id);g.fullSourceTextOverlap=g.cases.filter(c=>['hero-source','owner-mechanism'].includes(c.task)&&sourceTrain.has(digest(JSON.parse(c.messages[1].content).source.text))).map(c=>c.id);if(g.name.startsWith('core-')){assert.equal(g.exactPublicTrainOverlap.length,0);assert.equal(g.fullSourceTextOverlap.length,0);}}
 const pinPaths=[self,...['nonllm-baseline.mjs','nonllm-baseline-run.mjs','r6-score.mjs','r3-score.mjs','dataset.mjs','composition-client.mjs'].map(n=>path.join(dir,n)),...['selection.json','native-reference.json','experiment-policy.json','training-run.json','checkpoint-archive/training-order.json','cases.private.json','dataset-manifest.json'].map(n=>path.join(root,n)),...['cases.private.json','selection.json','native-reference.json'].map(n=>path.join(e.parent,n)),...groups.map(g=>g.file)];const pins=[...new Set(pinPaths)].map(p=>({path:p,sha256:fileHash(p)}));fs.mkdirSync(out);const put=(n,v)=>fs.writeFileSync(path.join(out,n),JSON.stringify(v,null,2)+'\n',{flag:'wx'});
 put('EXPOSURE.json',{...e,train:undefined,trainingRows:e.train.length});put('POLICY.json',{createdAt:new Date().toISOString(),root,pins,algorithmUnchanged:true,kGrid,thresholdGrid,devRows:158,groups:groups.map(({cases,...g})=>({...g,count:cases.length})),selection:'Both accuracy-first and safety-first K/threshold from 158 dev only; store before any test/post prediction. No best-policy choice from test.',vfxBaselineNotEvaluated:'No VFX dev in this selected-core protocol; all model VFX regressions remain required and are NOT waived.',modelTrainingChanged:false,checkpointReselected:false,releaseQualified:false});
 const start=performance.now(),model=fitExamples(e.train),fitSeconds=(performance.now()-start)/1000,selection={nearest:selectDev(dev,dev.map(c=>model.neighbors(c.messages)),'nearest'),catalog:selectDev(dev,dev.map(c=>catalogRanking(c.messages)),'catalog')};put('SELECTION.json',selection);
 let observedMaxRssBytes=process.memoryUsage().rss;const reports=[];
 for(const g of groups){const byArm={};for(const c of g.cases){const start=performance.now(),near=model.neighbors(c.messages),catalog=catalogRanking(c.messages),prepareSeconds=(performance.now()-start)/1000;
   for(const kind of['nearest','catalog'])for(const mode of['accuracy','safety']){const parameter=selection[kind].policies[mode][c.task];if(parameter===undefined)continue;const t=performance.now(),p=kind==='nearest'?chooseNeighbors(near,parameter):chooseCatalog(catalog,parameter);assert(p);(byArm[kind+'-'+mode]??=[]).push({id:c.id,requestDigest:c.requestDigest,value:p.value,error:null,seconds:prepareSeconds+(performance.now()-t)/1000,parameter,trace:kind==='nearest'?p.nearest:p.top});}observedMaxRssBytes=Math.max(observedMaxRssBytes,process.memoryUsage().rss);}
  const scores={};for(const[name,results]of Object.entries(byArm)){const ids=new Set(results.map(r=>r.id)),cs=g.cases.filter(c=>ids.has(c.id)),raw={complete:true,metadata:{backend:'node-tfidf',notLLM:true},results};put(g.name+'-'+name+'.json',raw);scores[name]=summarize(cs,raw);}
  put(g.name+'-scores.json',scores);reports.push({name:g.name,scores:Object.fromEntries(Object.entries(scores).map(([n,s])=>[n,{total:s.total,passed:s.passed,unsafe:s.unsafe,tasks:s.tasks}]))});
 }
 for(const p of pins)assert.equal(fileHash(p.path),p.sha256,'PIN_CHANGED');const summary={completedAt:new Date().toISOString(),selectedStep:e.selectedStep,selectedAdapterSha256:e.adapterSha256,inheritedUniqueRows:434,newPrefixRows:e.selectedR7PrefixRows,rawTrainRows:model.trainingRows,uniqueTrainRows:model.uniqueRows,fitSeconds,observedMaxRssBytes,notTruePeakMemory:true,reports,modelPairedComparisonPending:true,historicalScoresRewritten:false,waveOracleStillFrozen:true,releaseQualified:false};put('summary.json',summary);return{out,selectedStep:e.selectedStep,rawTrainRows:model.trainingRows,uniqueTrainRows:model.uniqueRows,fitSeconds,groups:groups.length};
}
export function compareModels(baseline,out){assert([baseline,out].every(path.isAbsolute));assert(!fs.existsSync(out),'REFUSE_OVERWRITE');const policy=read(path.join(baseline,'POLICY.json')),root=policy.root,summary=read(path.join(baseline,'summary.json')),e=exposure(root);assert.equal(e.adapterSha256,summary.selectedAdapterSha256);assert.equal(read(path.join(root,'post-v1/state.json')).status,'complete-research-evidence');for(const p of policy.pins)assert.equal(fileHash(p.path),p.sha256,'PIN_CHANGED');const groups=datasets(root,e.parent),reports=[];fs.mkdirSync(out);const put=(n,v)=>fs.writeFileSync(path.join(out,n),JSON.stringify(v,null,2)+'\n',{flag:'wx'});
 for(const g of groups){const classical=read(path.join(baseline,g.name+'-scores.json')),scores={},comparisons=[];for(const name of['base','r3','selected']){const file=g.name==='core-dev'?path.join(root,name==='selected'?(e.selectedStep?'dev-step-'+e.selectedStep:'r3-dev')+'.json':name+'-dev.json'):g.name==='core-test'?path.join(root,name+'-test.json'):path.join(root,'post-v1',g.name+'-'+name+'.json'),raw=read(file),s=summarize(g.cases,raw);scores[name]={total:s.total,passed:s.passed,unsafe:s.unsafe,tasks:s.tasks,rawPath:file,rawSha256:fileHash(file),meanGenerationSeconds:raw.results.reduce((n,r)=>n+r.seconds,0)/raw.results.length};for(const[b,bs]of Object.entries(classical)){const ids=new Set(bs.rows.map(r=>r.id));comparisons.push({before:b,after:name,byTask:paired(bs,{...s,rows:s.rows.filter(r=>ids.has(r.id))})});}}
  const report={name:g.name,classical:Object.fromEntries(Object.entries(classical).map(([n,s])=>[n,{total:s.total,passed:s.passed,unsafe:s.unsafe,tasks:s.tasks}])),models:scores,comparisons,sameScorerAndPublicRequests:true,sameBackend:false};put(g.name+'-comparison.json',report);reports.push(report);
 }
 const result={completedAt:new Date().toISOString(),selectedStep:e.selectedStep,selectedAdapterSha256:e.adapterSha256,baseline,rawTrainRows:summary.rawTrainRows,uniqueTrainRows:summary.uniqueTrainRows,reports,postHoc:true,algorithmUnchanged:true,checkpointReselected:false,selectionByTest:false,releaseQualified:false};put('summary.json',result);const lines=['# R7選中權重的實際訓練暴露：非LLM對照','',`R3 epoch2的434個獨特train＋R7選中step${e.selectedStep}的${e.selectedR7PrefixRows}例前綴；原始${summary.rawTrainRows}，公開輸入／target去重${summary.uniqueTrainRows}。不是R7準備的全部466，也不是worker最後380。`,'','相同監督例可用不等於相同預訓練、梯度權重或演算法。TF-IDF特徵與K／threshold grid完全沿用前版，兩個dev選參數政策均報告；不按test挑有利政策。VFX非LLM未測，但模型VFX保留檢查仍必須完成。','','| 題組 | 任務 | 方法 | 正確 | 錯誤放行 |','| --- | --- | --- | ---: | ---: |'];for(const r of reports)for(const[n,s]of Object.entries({...r.classical,...r.models}))for(const[t,v]of Object.entries(s.tasks))lines.push(`| ${r.name} | ${t} | ${n} | ${v.passed}/${v.total} | ${v.unsafe} |`);lines.push('','全為既有回歸／已知家族，不冒充獨立Gold。此處保持R7原目錄／wave oracle，以同題比較；新版校正後582題另外評測，不用新答案重算舊輸出。','');fs.writeFileSync(path.join(out,'REPORT.md'),lines.join('\n'),{flag:'wx'});return{out,groups:reports.length,releaseQualified:false};
}
if(process.argv[1]&&path.resolve(process.argv[1])===self){const[mode,...args]=process.argv.slice(2);assert(['run','compare'].includes(mode));console.log(JSON.stringify(mode==='run'?run(...args):compareModels(...args)));}
