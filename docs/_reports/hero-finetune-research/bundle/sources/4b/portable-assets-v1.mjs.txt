/** Freeze portable public examples, corrected catalogs and private check oracles before GPU output. */
import fs from'node:fs';import path from'node:path';import assert from'node:assert/strict';import{fileURLToPath}from'node:url';import{digest}from'./dataset.mjs';import{fileHash}from'./precision-diagnostic.mjs';import{prepare}from'./research-inference-v2.mjs';import{correctedDescriptions,correctionId}from'./r8-data-v1.mjs';
const self=fileURLToPath(import.meta.url),read=p=>JSON.parse(fs.readFileSync(p));
export function create(singlePath,stackPath,vfxPath,out){
 assert([singlePath,stackPath,vfxPath,out].every(path.isAbsolute));assert(!fs.existsSync(out),'REFUSE_OVERWRITE');const single=structuredClone(read(singlePath)),stack=read(stackPath),vfx=read(vfxPath);assert.equal(stack.capabilityRevision,correctionId);
 single.previousReviewSha256=digest(single);single.reviewVersion++;single.capabilityRevision=correctionId;single.rows=single.rows.map(r=>Object.hasOwn(correctedDescriptions,r.templateId)?{...r,description:correctedDescriptions[r.templateId]}:r);single.releaseQualified=false;
 const catalogs={single,stack,vfx};fs.mkdirSync(out,{recursive:true});fs.mkdirSync(path.join(out,'catalogs'));fs.mkdirSync(path.join(out,'examples'));const put=(p,v)=>fs.writeFileSync(path.join(out,p),JSON.stringify(v,null,2)+'\n',{flag:'wx'});for(const[k,v]of Object.entries(catalogs))put('catalogs/'+k+'.json',v);
 const sourceText='這份GGD改編設定使用共通魔力，沒有全圖追蹤。普攻追加物理傷害有兩秒內置冷卻。指定單體技能施放當下立即回復自身生命。',source={id:'portable-authored-fixture',name:'公開操作範例，不是Owner Gold',version:'portable-example-v1',text:sourceText,sha256:digest(sourceText)};
 const spec=(kind,id,request)=>({kind,catalog:{path:'../catalogs/'+kind+'.json',sha256:digest(catalogs[kind])},input:kind==='vfx'?{id,task:'vfx',request}:{id,request}});
 const all=[
  {id:'source-supported',spec:{kind:'source',input:{id:'source-supported',task:'hero-source',source,claim:'本改編使用共通魔力，而且沒有全圖追蹤。'}},expected:[[{verdict:'supported'}]],reason:'All clauses are directly stated.'},
  {id:'source-unknown',spec:{kind:'source',input:{id:'source-unknown',task:'owner-mechanism',source,claim:'指定單體技能的魔力消耗固定為100。'}},expected:[[{verdict:'not-stated'}]],reason:'Resource exists, but its exact skill cost is unstated.'},
  {id:'source-timing',spec:{kind:'source',input:{id:'source-timing',task:'owner-mechanism',source,claim:'施放指定單體技能當下不回血，必須等十秒之後才回血。'}},expected:[[{verdict:'contradicted'}]],reason:'Explicit immediate heal contradicts delayed-only heal.'},
  {id:'single-damage',spec:spec('single','single-damage','指定一名敵人立即扣血一次，不位移、不附控制或持續傷害。'),expected:[[{decision:'accept',templateId:'tpl-single-strike'}],[{decision:'accept',templateId:'tpl-instant-blast'}],[{decision:'accept',templateId:'tpl-proxy-cast'}]],reason:'Enabled single-target immediate-damage alternatives are intentionally allowed after manual settings.'},
  {id:'single-independent-arrival',spec:spec('single','single-independent-arrival','一具模型向前行進，沿途掃到的敵人各受傷一次；抵達後獨立範圍爆炸，沿途已受傷但仍在爆炸圈的人要再次受傷。不要求任意獨立取樣節拍。'),expected:[[{decision:'accept',templateId:'tpl-line-blast'}]],reason:'Preserved two-seed runtime proof has independent arrival; current traveling-wave terminal is not equivalent.'},
  {id:'stack-two-hits',spec:spec('stack','stack-two-hits','一次對指定敵人施法，同次結算打出兩筆獨立單體傷害，不合成一筆，也不要延時或第二次按鍵。'),expected:[[{decision:'accept',templateIds:['tpl-single-strike','tpl-single-strike'],onConflict:'reject'}]],reason:'Exact reviewed two-strike plan; duplicate IDs must survive.'},
  {id:'stack-independent-hooks',spec:spec('stack','stack-independent-hooks','常駐被動：普攻命中時追加傷害給普攻目標，自己實際受傷後反擊該次攻擊者。兩事件使用各自獨立的機率及內置冷卻，不取消原傷害。'),expected:[[{decision:'accept',templateIds:['tpl-on-attack','tpl-on-hit-react'],onConflict:'reject'}],[{decision:'accept',templateIds:['tpl-on-hit-react','tpl-on-attack'],onConflict:'reject'}]],reason:'Exact reviewed permanent independent-hook plan; both listed orders are allowed.'},
  {id:'stack-temporary-hooks',spec:spec('stack','stack-temporary-hooks','按下技能後才暫時獲得普攻追加和受傷反擊，五秒後兩者都移除。不是常駐被動；不能把五秒限制省略。'),expected:[[{decision:'refuse',templateIds:[],onConflict:'reject'}]],reason:'Available independent hook plan is permanent, not skill-granted expiring hooks.'},
  {id:'vfx-ki',spec:spec('vfx','vfx-ki','只推薦既有氣功砲光束特效模板，不要特效參數或傷害機制。'),expected:[[{decision:'accept',suggestedTemplateIds:['fx.prim.ki.beam'],reasonCode:'matching-template'}]],reason:'Personally inspected public catalog: fx.prim.ki.beam is usable=true, element=ki, shape=beam, standard 氣功／長直光束. This checks named template classification, not visual quality.'},
  {id:'fidelity-full',spec:{kind:'fidelity',input:{task:'owner-mechanism',source,proposal:sourceText,requirementsSourceSha256:source.sha256,requirementsCoverage:'reviewed-complete',requirements:[{id:'resource',text:'使用共通魔力。'},{id:'no-global-track',text:'沒有全圖追蹤。'},{id:'attack-cooldown',text:'普攻追加物理傷害有兩秒內置冷卻。'},{id:'immediate-self-heal',text:'指定單體技能施放當下立即回復自身生命。'}]}},expected:[Array.from({length:5},()=>({verdict:'supported'}))],reason:'Entire authored source is the proposal; four explicit requirements are checked bidirectionally. This is not real Owner completeness proof.'},
 ];
 const evidence=[];for(const r of all){put('examples/'+r.id+'.json',r.spec);const absolute=structuredClone(r.spec);if(absolute.catalog)absolute.catalog.path=path.resolve(out,'examples',absolute.catalog.path);const p=prepare(absolute);put(r.id+'-prepared.json',p);evidence.push({id:r.id,kind:r.spec.kind,expectedValues:r.expected,semanticGoldAvailable:r.expected!==null,reason:r.reason,requests:p.requests.length});}
 put('EXPECTED.private.json',{frozenAt:new Date().toISOString(),examples:evidence,trainingAllowed:false,checkpointSelectionAllowed:false,sourceAuthority:'Assistant-authored public usage fixtures; not independent Owner Gold',releaseQualified:false});
 put('ASSET_RECEIPT.json',{createdAt:new Date().toISOString(),sourcePins:[self,singlePath,stackPath,vfxPath].map(p=>({path:p,sha256:fileHash(p)})),catalogs:Object.fromEntries(Object.entries(catalogs).map(([k,v])=>[k,{path:'catalogs/'+k+'.json',sha256:digest(v)}])),examples:all.map(r=>({id:r.id,path:'examples/'+r.id+'.json',sha256:fileHash(path.join(out,'examples',r.id+'.json'))})),privateFilesExcludedFromPortableBundle:['EXPECTED.private.json','*-prepared.json'],gpuCalls:0,releaseQualified:false});return{out,examples:all.length,requests:evidence.reduce((s,r)=>s+r.requests,0),semanticGoldExamples:evidence.filter(r=>r.semanticGoldAvailable).length,trainingAllowed:false};
}
if(process.argv[1]&&path.resolve(process.argv[1])===self)console.log(JSON.stringify(create(...process.argv.slice(2))));
