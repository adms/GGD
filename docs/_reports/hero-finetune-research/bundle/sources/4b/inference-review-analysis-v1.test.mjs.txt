import test from'node:test';import assert from'node:assert/strict';import{errorKind,tally,sourceUnits,orderKey,sourceMatrix}from'./inference-review-analysis-v1.mjs';
test('source hallucination and over-refusal remain separate from format errors and training accuracy',()=>{const c={task:'owner-mechanism',acceptedTargets:[{verdict:'not-stated'}]},r={value:{verdict:'supported'},error:null,seconds:.5,score:{pass:false,schema:true,unsafe:true}};assert.equal(errorKind(c,r),'unsupported-source-accepted');assert.equal(errorKind({...c,acceptedTargets:[{verdict:'supported'}]},{...r,value:{verdict:'not-stated'}}),'supported-source-rejected');assert.equal(errorKind(c,{...r,value:null,error:'JSON'}),'parse-or-generation-error');const s=tally([{case:c,task:c.task,outputs:{r3:r}}],'r3');assert.equal(s.total,1);assert.equal(s.passed,0);assert.equal(s.unsafe,1);assert.equal(s.meanGenerationSeconds,.5);assert.equal(tally([],'r3').percent,null);});
test('one failed claim prevents the whole source unit from passing',()=>{const row=(id,pass)=>({id,task:'hero-source',input:{source:{text:'one source'}},outputs:{r3:{score:{pass,unsafe:false}}}});const s=sourceUnits([row('a',true),row('b',false)],'r3');assert.equal(s.total,1);assert.equal(s.allProvidedClaimsPassed,0);assert.equal(s.completeOriginalRequirementsProven,false);});
test('source confusion preserves direction and keeps invalid outputs out of semantic verdicts',()=>{const row=(task,gold,pred,schema=true)=>({task,case:{target:{verdict:gold}},outputs:{r3:{error:null,value:{verdict:pred},score:{schema}}}}),s=sourceMatrix([row('hero-source','contradicted','not-stated'),row('hero-source','not-stated','contradicted'),row('hero-source','supported','supported',false),row('owner-mechanism','supported','supported')],'r3','hero-source');assert.equal(s.total,3);assert.deepEqual(s.matrix,{contradicted:{'not-stated':1},'not-stated':{contradicted:1},supported:{invalid:1}});});
test('catalog permutation audit does not erase card order or duplicate multiplicity',()=>{const make=p=>({messages:[{role:'system',content:'fixed'},{role:'user',content:JSON.stringify({allowedPlans:p})}]}),a={id:'a',templateIds:['x','x']},b={id:'b',templateIds:['y','z']};assert.equal(orderKey(make([a,b])),orderKey(make([b,a])));assert.notEqual(orderKey(make([a,b])),orderKey(make([{...a,templateIds:['x']},b])));assert.notEqual(orderKey(make([a,b])),orderKey(make([a,{...b,templateIds:['z','y']}])));});
