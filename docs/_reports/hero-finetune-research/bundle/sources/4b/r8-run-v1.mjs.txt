/** Final bounded native LoRA experiment. No automatic preparation, restart, export or activation. */
import fs from'node:fs';import path from'node:path';import assert from'node:assert/strict';import{spawn}from'node:child_process';import{fileURLToPath}from'node:url';import{fileHash}from'./precision-diagnostic.mjs';import{verify}from'./r8-verify-v1.mjs';import{summarize,preservation,rank,eligible,compare,render}from'./r6-score.mjs';
const self=fileURLToPath(import.meta.url),dir=path.dirname(self),read=p=>JSON.parse(fs.readFileSync(p));
async function main(root,python,runtime){
 assert(!fs.existsSync(path.join(root,'TRAINING_DISABLED.json')),'USER_STOPPED_NEW_TRAINING');
 assert([root,python,runtime].every(path.isAbsolute));assert(!fs.existsSync(path.join(root,'run-state.json')),'REFUSE_RESTART');verify(root);const prep=read(path.join(root,'RUN_PREPARATION.json')),policyPath=path.join(root,'experiment-policy.json'),policy=read(policyPath);assert.equal(fileHash(policyPath),prep.policySha256);for(const p of prep.pins)assert.equal(fileHash(p.path),p.sha256,'PIN_CHANGED');assert.equal(policy.experiment,'r8-final-source-rehearsal-and-capability-correction');assert.equal(policy.trainRows,550);assert.equal(policy.trainingEpochs,1);assert.equal(policy.learningRate,2.5e-6);assert(Date.now()<Date.parse(policy.trainingCutoff)-30000,'START_CUTOFF');
 for(const p of prep.predecessors){const s=read(p.path);assert.equal(s.status,p.status);assert.equal(s.pid,p.pid);let alive=false;try{process.kill(s.pid,0);alive=true;}catch(e){assert.equal(e.code,'ESRCH','PROCESS_STATE_UNCERTAIN');}assert(!alive,'PREDECESSOR_STILL_LIVE');}
 const initial=prep.initialChoice.reference,control=prep.r3Control,base=initial.base,cases=read(path.join(root,'cases.private.json')),rows=s=>cases.filter(c=>c.split===s),put=(n,v)=>{const p=path.join(root,n);fs.writeFileSync(p+'.tmp',JSON.stringify(v,null,2)+'\n');fs.renameSync(p+'.tmp',p);};assert.equal(fileHash(path.join(initial.adapter,'adapters.safetensors')),policy.initialAdapterSha256);
 const receiptPath=path.join(path.dirname(base),'NATIVE_REFERENCE.json'),receipt=read(receiptPath),baseFiles=receipt.files.filter(f=>f.name.startsWith('base/'));assert.equal(baseFiles.length,13);assert.equal(receipt.baseRevision,'851bf6e806efd8d0a36b00ddf55e13ccb7b8cd0a');
 const pins=[...prep.pins,...[self,policyPath,path.join(root,'RUN_PREPARATION.json'),receiptPath,path.join(control.adapter,'adapters.safetensors'),path.join(control.adapter,'adapter_config.json')].map(p=>({path:p,sha256:fileHash(p)}))],lock=path.join(runtime,'gpu.lock'),cutoff=Date.parse(policy.deadline)-policy.reportReserveSeconds*1000;
 fs.writeFileSync(lock,JSON.stringify({pid:process.pid,kind:'r8-source-rehearsal',root}),{flag:'wx'});const state={status:'running',pid:process.pid,startedAt:new Date().toISOString(),stages:[]};put('run-state.json',state);put('run-pins.json',{pins,base,initialAdapter:initial.adapter,initialAdapterSha256:initial.adapterSha256,policySha256:fileHash(policyPath),releaseQualified:false});let child=null,reason=null,killTimer;
 function guard(){assert(!reason,reason);assert(Date.now()<cutoff,'GPU_CUTOFF');for(const p of[root,prep.parent,read(path.join(root,'DATA_PREPARATION.json')).extension])assert(!fs.existsSync(path.join(p,'CANCEL')),'CANCEL');for(const p of pins)assert.equal(fileHash(p.path),p.sha256,'PIN_CHANGED:'+p.path);}
 function stop(why){reason??=why;if(child?.pid){try{process.kill(-child.pid,'SIGTERM');}catch{}killTimer=setTimeout(()=>{try{process.kill(-child.pid,'SIGKILL');}catch{}},3000);}}
 process.on('SIGTERM',()=>stop('SIGNAL'));process.on('SIGINT',()=>stop('SIGNAL'));
 function verifyBase(){guard();for(const f of baseFiles){const p=path.join(path.dirname(base),f.name);assert.equal(fs.statSync(p).size,f.bytes);assert.equal(fileHash(p),f.sha256,'BASE_CHANGED');}guard();return{checkedAt:new Date().toISOString(),baseRevision:receipt.baseRevision,files:baseFiles};}
 async function execute(stage,args){guard();state.stage=stage;put('run-state.json',state);const log=fs.openSync(path.join(root,stage+'.log'),'wx');child=spawn(python,args,{stdio:['ignore',log,log],detached:true});fs.closeSync(log);put('worker-lease.json',{pid:child.pid,parent:process.pid,stage});const timer=setInterval(()=>{try{guard();if(stage==='train')assert(Date.now()<Date.parse(policy.trainingHardStopAt),'TRAINING_HARD_STOP');}catch(e){stop(String(e));}},2000);let code;try{code=await new Promise((res,rej)=>{child.once('exit',res);child.once('error',rej);});}finally{clearInterval(timer);clearTimeout(killTimer);child=null;put('worker-lease.json',{pid:null,stage});}assert(!reason&&code===0,reason??'WORKER_EXIT:'+stage+':'+code);guard();state.stages.push({stage,completedAt:new Date().toISOString()});put('run-state.json',state);console.log(JSON.stringify(state.stages.at(-1)));}
 async function evaluate(stage,split,adapter){const output=path.join(root,stage+'.json'),args=[path.join(dir,'worker.py'),'eval','--model',base,'--policy',policyPath,'--data',path.join(root,split+'-requests.json'),'--output',output];if(adapter)args.push('--adapter',adapter);await execute(stage,args);const raw=read(output);assert.equal(raw.metadata.modelPath,base);assert.equal(raw.metadata.adapter,adapter??null);const score=summarize(rows(split),raw);put(stage+'-score.json',score);return{raw,score};}
 try{
  put('base-before.json',verifyBase());const baseDev=await evaluate('base-dev','dev',null),controlDev=await evaluate('r3-dev','dev',control.adapter),initialDev=initial.adapterSha256===control.adapterSha256?controlDev:await evaluate('initial-dev','dev',initial.adapter);
  const candidates=[{step:0,label:'unchanged-r3-control',origin:'r3-control',adapter:control.adapter,adapterSha256:control.adapterSha256,score:controlDev.score,preservation:preservation(controlDev.score,controlDev.score)}];if(initial.adapterSha256!==control.adapterSha256)candidates.push({step:0,label:'unchanged-r7-selected-control',origin:'r7-control',adapter:initial.adapter,adapterSha256:initial.adapterSha256,score:initialDev.score,preservation:preservation(controlDev.score,initialDev.score)});
  put('baseline-dev-report.json',compare(rows('dev'),[{name:'base',raw:baseDev.raw},{name:'r3',raw:controlDev.raw},{name:'initial',raw:initialDev.raw}]));
  await execute('doctor',[path.join(dir,'worker.py'),'doctor','--model',base,'--adapter',initial.adapter,'--policy',policyPath,'--data',path.join(root,'train.jsonl'),'--output',path.join(root,'doctor.json')]);
  const adapterRoot=path.join(runtime,path.basename(root)+'-adapter');await execute('train',[path.join(dir,'r6-train.py'),'--model',base,'--adapter',initial.adapter,'--policy',policyPath,'--data',path.join(root,'train.jsonl'),'--output',adapterRoot]);const training=read(path.join(adapterRoot,'training-run.json'));assert(training.restoredAllTrainableValuesExactly);assert.equal(training.preparedExamples,550);assert.equal(training.initialAdapterSha256,initial.adapterSha256);put('training-run.json',training);
  const order=read(path.join(adapterRoot,'training-order.json')),metrics=fs.readFileSync(path.join(adapterRoot,'metrics.jsonl'),'utf8').trim().split('\n').map(JSON.parse);assert.deepEqual(order.slice().sort(),rows('train').map(c=>c.id).sort());for(const[i,m]of metrics.entries()){assert.equal(m.example,i+1);assert.equal(m.id,order[i]);assert(Number.isFinite(m.loss));}
  const archive=path.join(root,'checkpoint-archive');fs.cpSync(adapterRoot,archive,{recursive:true,errorOnExist:true,force:false});for(const cp of training.checkpoints){assert.equal(fileHash(path.join(archive,'step-'+cp.step,'adapters.safetensors')),cp.sha256);assert.equal(metrics[cp.examplesSeen-1].committedExamples,cp.examplesSeen);assert.equal(metrics[cp.examplesSeen-1].step,cp.step);}
  const last=training.checkpoints.at(-1),selectedChecks=training.checkpoints.filter(c=>policy.evaluateSteps.includes(c.step)||c.step===last.step);assert(selectedChecks.length>0&&selectedChecks.length<=4);
  for(const cp of selectedChecks){const adapter=path.join(archive,'step-'+cp.step),result=await evaluate('dev-step-'+cp.step,'dev',adapter);candidates.push({step:cp.step,label:'r8-step-'+cp.step,origin:'r8-trained',adapter,adapterSha256:cp.sha256,score:result.score,preservation:preservation(controlDev.score,result.score)});}
  candidates.sort(rank);const selected=candidates[0];assert(selected.preservation.pass);put('selection.json',{selected,allCandidates:candidates,selectedAt:new Date().toISOString(),testUsed:false,selectionUses:policy.selection,eligible:eligible(selected.score),trainingComplete:training.trainingComplete,newTrainingSelected:selected.origin==='r8-trained'});
  const keep=path.join(root,'selected-adapter');fs.mkdirSync(keep);for(const n of['adapters.safetensors','adapter_config.json'])fs.copyFileSync(path.join(selected.adapter,n),path.join(keep,n),fs.constants.COPYFILE_EXCL);assert.equal(fileHash(path.join(keep,'adapters.safetensors')),selected.adapterSha256);put('native-reference.json',{base,adapter:keep,adapterSha256:selected.adapterSha256,selectedStep:selected.step,selectedOrigin:selected.origin,newTrainingSelected:selected.origin==='r8-trained',trainingComplete:training.trainingComplete,decoding:{presencePenalty:0,thinking:false},releaseQualified:false});
  const selectedDev=selected.origin==='r3-control'?controlDev.raw:selected.origin==='r7-control'?initialDev.raw:read(path.join(root,'dev-step-'+selected.step+'.json'));const devReport=compare(rows('dev'),[{name:'base',raw:baseDev.raw},{name:'r3',raw:controlDev.raw},{name:'selected',raw:selectedDev}]);put('dev-report.json',devReport);fs.writeFileSync(path.join(root,'DEV_REPORT.md'),render(devReport),{flag:'wx'});
  const testArms=[];for(const[name,adapter]of[['base',null],['r3',control.adapter],['selected',keep]])testArms.push({name,raw:(await evaluate(name+'-test','test',adapter)).raw});const testReport=compare(rows('test'),testArms);put('test-report.json',testReport);fs.writeFileSync(path.join(root,'TEST_REPORT.md'),render(testReport),{flag:'wx'});put('base-after.json',verifyBase());put('comparison.json',{selectedStep:selected.step,selectedOrigin:selected.origin,newTrainingSelected:selected.origin==='r8-trained',trainingComplete:training.trainingComplete,devEligible:eligible(selected.score),testEligible:eligible(testReport.scores.selected),testScores:Object.fromEntries(Object.entries(testReport.scores).map(([n,s])=>[n,{total:s.total,passed:s.passed,unsafe:s.unsafe,tasks:s.tasks}])),requiredBroaderAndPortableChecksPending:true,releaseQualified:false,activation:false});state.status='complete-research-only';state.completedAt=new Date().toISOString();put('run-state.json',state);console.log(JSON.stringify({status:state.status,selectedOrigin:selected.origin,selectedStep:selected.step,trainingComplete:training.trainingComplete,releaseQualified:false}));
 }catch(e){state.status='failed';state.error=String(e);put('run-state.json',state);console.error(e);process.exitCode=1;}
 finally{clearTimeout(killTimer);if(fs.existsSync(lock)&&read(lock).pid===process.pid)fs.unlinkSync(lock);put('worker-lease.json',{pid:null,reason,endedAt:new Date().toISOString()});}
}
if(process.argv[1]&&path.resolve(process.argv[1])===self)await main(...process.argv.slice(2));
