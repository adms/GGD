/** Durable native-only research fallback. No fusion, quantization, activation or training. */
import fs from'node:fs';import path from'node:path';import assert from'node:assert/strict';import{fileURLToPath}from'node:url';import{reference}from'./final-reference-v1.mjs';import{copyFlat,inventory}from'./final-bundle-v1.mjs';import{fileHash}from'./precision-diagnostic.mjs';import{digest}from'./dataset.mjs';import{summarize}from'./r6-score.mjs';
const self=fileURLToPath(import.meta.url),read=p=>JSON.parse(fs.readFileSync(p));
export function build(referenceRoot,qualityRoot,arm,kit,assets,out){
 assert([referenceRoot,qualityRoot,kit,assets,out].every(path.isAbsolute));assert(['r3','selected'].includes(arm));assert(!fs.existsSync(out)&&!fs.existsSync(out+'.build.json'),'REFUSE_OVERWRITE');const ref=reference(referenceRoot),native=ref.native,rawPath=path.join(qualityRoot,arm+'-test.json'),raw=read(rawPath),cases=read(path.join(qualityRoot,'cases.private.json')).filter(c=>c.split==='test'),policyPath=path.join(qualityRoot,'experiment-policy.json'),policy=read(policyPath);assert.equal(raw.metadata.modelPath,native.base);assert.equal(raw.metadata.thinking,false);assert.equal(fileHash(path.join(raw.metadata.adapter,'adapters.safetensors')),native.adapterSha256);assert.equal(policy.presencePenalty,0);assert.equal(policy.temperature,0);assert.equal(policy.maxSequenceTokens,4096);assert.equal(policy.maxOutputTokens,256);const score=summarize(cases,raw);
 const base=read(path.join(path.dirname(native.base),'NATIVE_REFERENCE.json')),baseFiles=base.files.filter(f=>f.name.startsWith('base/'));assert.equal(baseFiles.length,13);assert.equal(base.baseRevision,'851bf6e806efd8d0a36b00ddf55e13ccb7b8cd0a');for(const f of baseFiles)assert.equal(fileHash(path.join(path.dirname(native.base),f.name)),f.sha256,'BASE_CHANGED');
 const code=read(path.join(kit,'CODE_RECEIPT.json')),a=read(path.join(assets,'ASSET_RECEIPT.json'));assert.equal(code.privateTrainingDataIncluded,false);for(const f of code.files)assert.equal(fileHash(path.join(kit,f.path)),f.sha256);for(const p of a.sourcePins)assert.equal(fileHash(p.path),p.sha256);for(const v of Object.values(a.catalogs))assert.equal(digest(read(path.join(assets,v.path))),v.sha256);for(const e of a.examples)assert.equal(fileHash(path.join(assets,e.path)),e.sha256);
 const state={status:'copying',pid:process.pid,startedAt:new Date().toISOString(),out,referenceRoot,qualityRoot,arm,releaseQualified:false};const record=()=>{fs.writeFileSync(out+'.build.json.tmp',JSON.stringify(state,null,2)+'\n');fs.renameSync(out+'.build.json.tmp',out+'.build.json');};record();
 try{fs.mkdirSync(out);copyFlat(path.join(kit,'code'),path.join(out,'code'));copyFlat(path.join(assets,'catalogs'),path.join(out,'catalogs'));copyFlat(path.join(assets,'examples'),path.join(out,'examples'));copyFlat(native.base,path.join(out,'models/native'));copyFlat(native.adapter,path.join(out,'models/native-adapter'));fs.copyFileSync(path.join(native.base,'LICENSE'),path.join(out,'LICENSE'),fs.constants.COPYFILE_EXCL);
  const put=(n,v)=>fs.writeFileSync(path.join(out,n),JSON.stringify(v,null,2)+'\n',{flag:'wx'});put('PROVENANCE.json',{baseModel:'Qwen/Qwen3.5-4B',baseRevision:base.baseRevision,selectedAdapterSha256:native.adapterSha256,sourceTrainingRun:path.basename(referenceRoot),referenceProtocol:ref.protocol,originalSelectionDecode:ref.historicalDecode,entryDecode:0,qualityMeasurement:{run:path.basename(qualityRoot),arm,rawSha256:fileHash(rawPath),caseSha256:fileHash(path.join(qualityRoot,'cases.private.json')),policySha256:fileHash(policyPath),testRows:score.total},code:code.files.map(({path:p,sha256,sourceSha256,mode,exports})=>({path:p,sha256,sourceSha256,mode,exports})),dataFilesNotDistributed:true,weightMemorizationNotAudited:true,quantizationPerformed:false,releaseQualified:false});
  fs.writeFileSync(path.join(out,'MODEL_CARD.md'),['# GGD來源／技能分類：native研究保存包','','這是可搬移的研究權重保存包，**不是穩定全用途模型或正式產品**。只做來源分類與既有模板推薦，不調VFX參數、不作視覺驗證，不自動改寫技能或啟用Editor。','','- Base：Qwen/Qwen3.5-4B，revision `'+base.baseRevision+'`。','- 原選模：'+JSON.stringify(ref.protocol)+'；原選定權重未更換。','- 本包只有native BF16 base＋LoRA，沒有做融合或量化。','- 公開入口固定non-thinking、presence penalty=0；原始歷史1.5分數不混算。','- 三種精度互相比較尚未在本包執行；缺少8bit不是8bit合格或不合格的證據。','','## 同協定既有核心回歸','','| 任務 | 正確 | 錯誤放行 |','| --- | ---: | ---: |',...Object.entries(score.tasks).map(([t,s])=>`| ${t} | ${s.passed}/${s.total} | ${s.unsafe} |`),'','這些是已曝光的研究回歸、非獨立Owner Gold；不是所有英雄設定／技能需求都已涵蓋。多模板與原文仍有實際錯誤，格式合法不能取代語意審查。目錄範例另採修正過的能力版本；上述核心數字不等於新目錄整體品質。','','## 使用限制','','本包建立時尚未從搬移後入口做GPU推論；須另看同名portable-check證據。入口配置12GiB上限不代表16GB MacBook實機已測；整機還需OS、Editor和其他程式記憶體。原測量只在本機M5 Max 128GB。','','保留原始LICENSE。權重經本專案分類資料微調。本包不附訓練／測試資料或私人答案檔案；不代表模型權重完全不記憶訓練內容。',''].join('\n'),{flag:'wx'});
  fs.writeFileSync(path.join(out,'README.md'),['# Mac研究模型使用','','先讀MODEL_CARD.md。需要Apple Silicon、arm64 Python 3.11及Node（此研究測25.9.0）。安裝會下載依賴；推論強制離線。','','在本資料夾：','','```sh','python3.11 -m venv .venv','.venv/bin/python -m pip install -r code/requirements.lock.txt','mkdir -p runtime','node code/run.mjs examples/source-supported.json .venv/bin/python runtime run-001 native RESEARCH_ONLY','```','','輸出目錄必須不存在。GPU鎖存在即拒絕，不終止其他工作；每次最多300秒、4096總token、256輸出token。來源過長報錯，不截斷。','','examples有source、fidelity、single、stack、vfx五種既有契約。fidelity只檢查呼叫者明列的requirement，不保證未列出的原文條件也完整涵蓋。VFX僅推薦模板，人工微調。','','source文字修改後，source.sha256必須同步更新。使用本包相同digest算法，例如：','','```sh','node --input-type=module -e \'import {digest} from "./code/dataset.mjs"; console.log(digest("新的完整原文"));\'','```','','只檢查hash與schema不等於語意正确。保留raw.json與checked.json，所有輸出均需人工核對來源與能力；不可自動執行或直接寫入遊戲。','','模型、tokenizer、config、adapter及程式均在MODEL_BUNDLE.json逐檔SHA256清單內。可搬移整個資料夾，不能只搬權重檔。',''].join('\n'),{flag:'wx'});
  const files=inventory(out);assert(!files.some(f=>/EXPECTED|cases\.private|train\.jsonl|owner-review|hero-review|prepared\.json/.test(f.path)));put('MODEL_BUNDLE.json',{schema:'ggd-mac-model-bundle@1',createdAt:new Date().toISOString(),baseModel:'Qwen/Qwen3.5-4B',baseRevision:base.baseRevision,adapterSha256:native.adapterSha256,defaultVariant:'native',variants:{native:{model:'models/native',adapter:'models/native-adapter',qualityQualified:false}},thinking:false,decoding:{presencePenalty:0,temperature:0,topP:.8,topK:20,maxSequenceTokens:4096,maxOutputTokens:256},files,quantizationPerformed:false,releaseQualified:false,physical16GBMachineTested:false,portableGpuEntryTested:false});state.status='copied-and-byte-verified';state.completedAt=new Date().toISOString();state.totalBytes=files.reduce((s,f)=>s+f.bytes,0);record();return{out,files:files.length,totalBytes:state.totalBytes,portableGpuEntryTested:false,releaseQualified:false};
 }catch(e){state.status='failed';state.error=String(e);record();throw e;}
}
if(process.argv[1]&&path.resolve(process.argv[1])===self)console.log(JSON.stringify(build(...process.argv.slice(2))));
