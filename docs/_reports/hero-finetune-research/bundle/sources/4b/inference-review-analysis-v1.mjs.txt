/** Re-score completed paired chunks, preserving exposure strata and unmeasured denominators. */
import fs from'node:fs';import path from'node:path';import assert from'node:assert/strict';import{fileURLToPath}from'node:url';import{fileHash}from'./precision-diagnostic.mjs';import{compare}from'./r6-score.mjs';import{digest}from'./dataset.mjs';
const self=fileURLToPath(import.meta.url),read=p=>JSON.parse(fs.readFileSync(p)),arms=['base','r3','r7'];
export function errorKind(c,r){if(r.error||!r.value)return'parse-or-generation-error';if(!r.score.schema)return'contract-invalid';if(r.score.pass)return'correct';const expected=c.acceptedTargets??[c.target],v=r.value;if(['hero-source','owner-mechanism'].includes(c.task)){if(v.verdict==='supported')return'unsupported-source-accepted';if(expected.some(x=>x.verdict==='supported'))return'supported-source-rejected';return'contradicted-vs-not-stated';}if(v.decision==='accept'&&expected.every(x=>x.decision==='refuse'))return'unsupported-request-accepted';if(v.decision==='refuse'&&expected.some(x=>x.decision==='accept'))return'valid-request-refused';return'wrong-template-or-plan';}
export function tally(rows,arm){const t={total:rows.length,passed:0,unsafe:0,schema:0,seconds:0,errors:{},tasks:{}};for(const r of rows){const o=r.outputs[arm],s=o.score,k=errorKind(r.case,o);t.passed+=+s.pass;t.unsafe+=+s.unsafe;t.schema+=+s.schema;t.seconds+=o.seconds;t.errors[k]=(t.errors[k]??0)+1;const task=t.tasks[r.task]??={total:0,passed:0,unsafe:0};task.total++;task.passed+=+s.pass;task.unsafe+=+s.unsafe;}t.percent=t.total?100*t.passed/t.total:null;t.meanGenerationSeconds=t.total?t.seconds/t.total:null;return t;}
export function sourceUnits(rows,arm){const units=new Map;for(const r of rows){if(!['hero-source','owner-mechanism'].includes(r.task))continue;const key=digest({task:r.task,text:r.input.source.text}),u=units.get(key)??{sourceSha256:key,task:r.task,ids:[],allPass:true,anyUnsafe:false};u.ids.push(r.id);u.allPass&&=r.outputs[arm].score.pass;u.anyUnsafe||=r.outputs[arm].score.unsafe;units.set(key,u);}const values=[...units.values()];return{total:values.length,allProvidedClaimsPassed:values.filter(u=>u.allPass).length,anyUnsafe:values.filter(u=>u.anyUnsafe).length,units:values,completeOriginalRequirementsProven:false};}
export function orderKey(c){const input=JSON.parse(c.messages[1].content);for(const k of['catalog','allowedPlans','unavailable'])if(Array.isArray(input[k]))input[k].sort((a,b)=>digest(a).localeCompare(digest(b)));return digest([c.messages[0],input]);}
export function sourceMatrix(rows,arm,task){const matrix={},selected=rows.filter(r=>r.task===task);for(const r of selected){const gold=[...new Set((r.case.acceptedTargets??[r.case.target]).map(t=>t.verdict))].sort().join('|'),o=r.outputs[arm],predicted=o.error||!o.score.schema?'invalid':o.value.verdict;(matrix[gold]??={})[predicted]=((matrix[gold]??{})[predicted]??0)+1;}return{total:selected.length,matrix};}
export function analyze(run,out){assert([run,out].every(path.isAbsolute));assert(!fs.existsSync(out),'REFUSE_OVERWRITE');const prep=read(path.join(run,'PREPARATION.json')),data=prep.data,manifest=read(path.join(data,'manifest.json')),state=read(path.join(run,'state.json')),cases=read(path.join(data,'cases.private.json')),mapping=read(path.join(data,'mapping.private.json')),byCase=new Map(cases.map(c=>[c.id,c])),byMap=new Map(mapping.map(c=>[c.id,c])),pins=[self,path.join(run,'PREPARATION.json'),...['manifest.json','cases.private.json','mapping.private.json'].map(n=>path.join(data,n))].map(p=>({path:p,sha256:fileHash(p)})),rows=[],hardware=[];
 for(const p of prep.pins)assert.equal(fileHash(p.path),p.sha256,'RUN_PIN_CHANGED');for(const done of state.pairedChunks){const chunk=manifest.chunks.find(c=>c.name===done.name);assert(chunk);const cs=chunk.ids.map(id=>byCase.get(id)),rawArms=arms.map(name=>{const p=path.join(run,chunk.name+'-'+name+'.json'),raw=read(p);pins.push({path:p,sha256:fileHash(p)});assert.equal(raw.metadata.adapter,name==='base'?null:prep.modelReferences[name].adapter);hardware.push({chunk:chunk.name,arm:name,peakMetalBytes:raw.metadata.peakMetalBytes,loadSeconds:raw.metadata.loadSeconds,rows:raw.results.length});return{name,raw};}),replayed=compare(cs,rawArms);for(const r of replayed.rows)rows.push({...r,case:byCase.get(r.id),mapping:byMap.get(r.id)});}
 const qualified=rows.filter(r=>r.mapping.scoringQualified),scores=Object.fromEntries(arms.map(a=>[a,tally(qualified,a)])),exposure={};for(const reference of['r3','r7']){exposure[reference]={};for(const bucket of['exact-training-input','new-input-seen-source','new-input-new-source','new-nonsource-input']){const rs=qualified.filter(r=>{const e=r.mapping.exposure[reference];return bucket==='exact-training-input'?e.exactTrainingInputSeen:bucket==='new-input-seen-source'?!e.exactTrainingInputSeen&&e.fullSourceTextSeen===true:bucket==='new-input-new-source'?!e.exactTrainingInputSeen&&e.fullSourceTextSeen===false:!e.exactTrainingInputSeen&&e.fullSourceTextSeen===null;});exposure[reference][bucket]=Object.fromEntries(arms.map(a=>[a,tally(rs,a)]));}}
 const datasetReports=manifest.cohorts.map(g=>{const planned=mapping.filter(m=>m.members.some(x=>x.dataset===g.name)),rs=qualified.filter(r=>r.mapping.members.some(m=>m.dataset===g.name));return{name:g.name,originalRows:g.rows,plannedUniqueRequests:planned.length,pairedQualifiedRequests:rs.length,pairedRequests:rows.filter(r=>r.mapping.members.some(m=>m.dataset===g.name)).length,scores:Object.fromEntries(arms.map(a=>[a,tally(rs,a)]))};});
 const community=r=>r.mapping.members.some(m=>m.id.includes('community-')),newHeroes=qualified.filter(community),communitySplits=Object.fromEntries(['train','dev','test'].map(s=>[s,Object.fromEntries(arms.map(a=>[a,tally(newHeroes.filter(r=>r.mapping.members.some(m=>m.id.includes('community-')&&m.split===s)),a)]))]));const transitions=[];for(const before of arms)for(const after of arms){if(arms.indexOf(before)>=arms.indexOf(after))continue;const improved=qualified.filter(r=>!r.outputs[before].score.pass&&r.outputs[after].score.pass),regressed=qualified.filter(r=>r.outputs[before].score.pass&&!r.outputs[after].score.pass),newUnsafe=qualified.filter(r=>!r.outputs[before].score.unsafe&&r.outputs[after].score.unsafe);transitions.push({before,after,improved:improved.map(r=>r.id),regressed:regressed.map(r=>r.id),newUnsafe:newUnsafe.map(r=>r.id)});}
 const failures=qualified.filter(r=>arms.some(a=>!r.outputs[a].score.pass)).map(r=>({...r,errorKinds:Object.fromEntries(arms.map(a=>[a,errorKind(r.case,r.outputs[a])]))}));failures.sort((a,b)=>Number(community(b))-Number(community(a))||Number(['hero-source','owner-mechanism'].includes(b.task))-Number(['hero-source','owner-mechanism'].includes(a.task))||Number(b.outputs.base.score.pass&&!b.outputs.r3.score.pass)-Number(a.outputs.base.score.pass&&!a.outputs.r3.score.pass)||a.id.localeCompare(b.id));
 const result={analyzedAt:new Date().toISOString(),run,runStatus:state.status,runError:state.error??null,plannedUniqueRequests:cases.length,pairedRequests:rows.length,pairedQualifiedRequests:qualified.length,excludedScoringRows:rows.filter(r=>!r.mapping.scoringQualified).map(r=>({id:r.id,mapping:r.mapping})),notPaired:cases.filter(c=>!rows.some(r=>r.id===c.id)).map(c=>({id:c.id,task:c.task})),scores,exposure,datasetReports,community:{planned:147,paired:newHeroes.length,trained:false,scores:Object.fromEntries(arms.map(a=>[a,tally(newHeroes,a)])),splits:communitySplits},transitions,hardware,pins,trainingCalls:0,checkpointReselected:false,independentOwnerGold:false,releaseQualified:false,scope:'All scores are unique complete paired requests with non-withdrawn labels. Historical contexts are not current-engine certification. Training input performance is not generalization; unresolved data quality is not automatically model failure.'};
 result.community.sourceUnits=Object.fromEntries(arms.map(a=>[a,sourceUnits(newHeroes,a)]));result.sourceUnits=Object.fromEntries(arms.map(a=>[a,sourceUnits(qualified,a)]));
 result.sourceConfusions=Object.fromEntries(['hero-source','owner-mechanism'].map(task=>[task,Object.fromEntries(arms.map(a=>[a,sourceMatrix(qualified,a,task)]))]));
 const orderGroups=new Map;for(const c of cases){const key=orderKey(c);if(!orderGroups.has(key))orderGroups.set(key,[]);orderGroups.get(key).push(c.id);}const completedById=new Map(qualified.map(r=>[r.id,r])),duplicates=[...orderGroups.entries()].filter(([,ids])=>ids.length>1),matched=duplicates.filter(([,ids])=>ids.every(id=>completedById.has(id))).map(([key,ids])=>({key,ids,sameAcceptedAnswers:new Set(ids.map(id=>digest(byCase.get(id).acceptedTargets.map(digest).sort()))).size===1,byArm:Object.fromEntries(arms.map(a=>[a,{sameValue:new Set(ids.map(id=>digest(completedById.get(id).outputs[a].value))).size===1,samePass:new Set(ids.map(id=>completedById.get(id).outputs[a].score.pass)).size===1,values:ids.map(id=>({id,value:completedById.get(id).outputs[a].value,pass:completedById.get(id).outputs[a].score.pass}))}]))}));
 result.catalogOrderAudit={exactRequests:cases.length,normalizedRequests:orderGroups.size,additionalOrderVariants:cases.length-orderGroups.size,plannedDuplicateGroups:duplicates.length,completedQualifiedGroups:matched.length,groups:matched,scope:'Only catalog/allowedPlans/unavailable list order is ignored for this diagnostic. System text, source, descriptions and order/multiplicity inside templateIds remain unchanged. All original requests are still evaluated. Index-derived seeds differ; temperature is zero. Not an independently randomized causal experiment.'};
 fs.mkdirSync(out);const put=(n,v)=>fs.writeFileSync(path.join(out,n),JSON.stringify(v,null,2)+'\n',{flag:'wx'});put('summary.json',result);put('all-failures.private.json',failures);put('review-priority.private.json',failures.slice(0,60));
 const lines=['# 全資料推論對照與錯誤分析','',`分析時間：${result.analyzedAt}；推論程序：${state.status}。計畫${cases.length}題，完整三模型配對${rows.length}題，可信評分分母${qualified.length}題；未配對${result.notPaired.length}題。`,'','沒有新增訓練，R8未訓練；不由這份分析重選checkpoint。版本盤點見資料目錄INVENTORY.md。','','## 按任務／模型','','| 任務 | 模型 | 正確 | 誤放行 |','| --- | --- | ---: | ---: |'];for(const a of arms)for(const[t,s]of Object.entries(scores[a].tasks))lines.push(`| ${t} | ${a} | ${s.passed}/${s.total} | ${s.unsafe} |`);
 lines.push('','## 七個新角色來源，未訓練','','這147題原本的train/dev/test標籤只代表預先固定的資料切分；任何一題都未用於本次新訓練。沒有獨立Owner Gold，仍屬經審查的GGD改編來源研究。','','| 原切分 | 模型 | 正確 | 誤放行 |','| --- | --- | ---: | ---: |');for(const[s,as]of Object.entries(communitySplits))for(const[a,t]of Object.entries(as))lines.push(`| ${s} | ${a} | ${t.passed}/${t.total} | ${t.unsafe} |`);

 lines.push('','## 版本覆蓋','','同一請求可能屬於多版，下面不能相加為獨特資料量。','','| 資料版 | 原始列 | 計畫獨特請求 | 已配對 | 有效評分 |','| --- | ---: | ---: | ---: | ---: |',...datasetReports.map(r=>`| ${r.name} | ${r.originalRows} | ${r.plannedUniqueRequests} | ${r.pairedRequests} | ${r.pairedQualifiedRequests} |`),'','## 目錄排列重複與穩定性','',`${cases.length}個完全訊息去重請求，若僅忽略catalog／allowedPlans／unavailable的列表排列，剩${orderGroups.size}個；${cases.length-orderGroups.size}個額外排列不是同樣數目的新語意需求。${matched.length}/${duplicates.length}組已完整配對，可在summary.json查看排列不同時答案／正確性是否改變。計畫內卡片順序、重複數、描述、原文與system完全保留，不為去重刪掉語意。`,'','## 判讀限制','','- 不是所有歷史隔離／舊proposal資料都納入此次新推論，未匹配項目逐份列於INVENTORY.md；不能把已排隊當完成。','- 新錯誤放行、過度拒絕、錯誤模板／計畫、來源contradicted與not-stated混淆分開計數，詳見summary.json。','- all-failures.private.json保留所有配對失敗的完整输入、預期、原始值與理由；review-priority.private.json優先列新七角色與來源退步供人工分析。自動錯誤分類不代表已逐題人工裁定。','- 12GiB配置只測這台M5 Max，非16GB MacBook實機證明。生成耗時不含全部檔案校驗或入口冷啟動。','- 全部維持research-only；平均分、格式合法或訓練題命中不等於完整英雄技能鑄造合格。','');fs.writeFileSync(path.join(out,'REPORT.md'),lines.join('\n'),{flag:'wx'});return{out,status:state.status,paired:rows.length,planned:cases.length,community:newHeroes.length};
