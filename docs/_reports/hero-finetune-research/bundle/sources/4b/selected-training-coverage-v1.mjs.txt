export function counts(rows){const tasks={},single={},stack={},vfx={},source={};for(const c of rows){tasks[c.task]=(tasks[c.task]??0)+1;const t=c.target;if(['hero-source','owner-mechanism'].includes(c.task)){source[c.task]??={};source[c.task][t.verdict]=(source[c.task][t.verdict]??0)+1;}if(t.decision!=='accept')continue;if(c.task==='mechanism-template')single[t.templateId]=(single[t.templateId]??0)+1;else if(c.task==='mechanism-stack'){const signature=JSON.stringify(t.templateIds);(stack[signature]??={templateIds:t.templateIds,rows:0,ids:[]}).rows++;stack[signature].ids.push(c.id);}else if(c.task==='vfx-semantic')for(const id of t.suggestedTemplateIds)vfx[id]=(vfx[id]??0)+1;}
 const multi=Object.values(stack).filter(s=>s.templateIds.length>1);return{rows:rows.length,tasks,sourceLabels:source,singleTeacherPositiveCounts:single,vfxTeacherPositiveCounts:vfx,stackTeacherPositiveSignatures:Object.values(stack),stackAcceptedRows:Object.values(stack).reduce((n,s)=>n+s.rows,0),actualMulticardPositiveRows:multi.reduce((n,s)=>n+s.rows,0),multicardSignatures:multi.length,repeatedCardPositiveRows:multi.filter(s=>new Set(s.templateIds).size<s.templateIds.length).reduce((n,s)=>n+s.rows,0),alternativeAcceptedTargetsNotCountedAsTeacherExposure:true};}
