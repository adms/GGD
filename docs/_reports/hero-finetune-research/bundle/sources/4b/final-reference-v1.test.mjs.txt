import test from'node:test';import assert from'node:assert/strict';import path from'node:path';import{fileURLToPath}from'node:url';import{selectionProtocol,reference}from'./final-reference-v1.mjs';
test('only known dev-only epoch/step protocols are accepted',()=>{assert.equal(selectionProtocol({testUsed:false,selected:{step:0}}).kind,'step-dev-only');assert.equal(selectionProtocol({selectionData:'dev-only; test not evaluated',selected:{epoch:2}}).kind,'r3-epoch-dev-only');assert.throws(()=>selectionProtocol({testUsed:true,selected:{step:48}}));assert.throws(()=>selectionProtocol({selectionData:'best test accuracy',selected:{epoch:2}}));});
test('previously completed R3 remains independently verifiable; new zero-penalty evaluation is required',()=>{const root=path.resolve(path.dirname(fileURLToPath(import.meta.url)),'../../../outputs/forge-mechanism-priority-r3-20260906'),r=reference(root);assert.equal(r.protocol.epoch,2);assert.equal(r.native.adapterSha256,'54ca2eb91c3c2313f1911ee4e7026571685776650f1505d2af67bc38bd436c06');assert.equal(r.historicalDecode,1.5);assert.equal(r.newEvaluationDecode,0);assert.equal(r.newEvaluationRequired,true);assert.equal(r.releaseQualified,false);});
